### 4.1 BPMN 流程图查看器（Vue Flow）

```vue
<!-- views/flow/FlowViewer.vue -->
<template>
  <div class="flow-viewer">
    <VueFlow :nodes="nodes" :edges="edges" :default-viewport="{ zoom: 0.8 }">
      <template #node-start="{ data }">
        <div class="bpmn-node start">○</div>
      </template>
      <template #node-end="{ data }">
        <div class="bpmn-node end">⊕</div>
      </template>
      <template #node-task="{ data }">
        <div class="bpmn-node task" :class="data.status">
          <div class="task-name">{{ data.label }}</div>
          <div class="task-actor">{{ data.actor }}</div>
        </div>
      </template>
      <template #node-gateway="{ data }">
        <div class="bpmn-node gateway">◇</div>
      </template>
      <template #node-user-task="{ data }">
        <div class="bpmn-node user-task" :class="data.status">
          <div class="task-name">👤 {{ data.label }}</div>
        </div>
      </template>
    </VueFlow>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted, computed } from 'vue'
import { VueFlow } from '@vue-flow/core'
import { getProcessGraph } from '../../sdk'

const props = defineProps<{ processId: string }>()

const graphData = ref(null)

onMounted(async () => {
  graphData.value = await getProcessGraph(props.processId)
})

const nodes = computed(() => {
  if (!graphData.value) return []
  return graphData.value.nodes.map(n => ({
    id: n.id,
    type: mapNodeType(n.type),
    position: { x: n.x, y: n.y },
    data: { label: n.label, actor: n.actor, status: n.status },
  }))
})

const edges = computed(() => {
  if (!graphData.value) return []
  return graphData.value.edges.map(e => ({
    id: e.id || `${e.source}-${e.target}`,
    source: e.source,
    target: e.target,
    label: e.label,
    animated: e.type === 'conditional',
  }))
})

function mapNodeType(type: string) {
  switch (type) {
    case 'event': return 'start'
    case 'end_event': return 'end'
    case 'activity': return 'task'
    case 'user_task': return 'user-task'
    case 'gateway': return 'gateway'
    default: return 'task'
  }
}
</script>
```

### 4.3 容器日志查看器

```vue
<!-- views/sandbox/ContainerLogs.vue -->
<template>
  <div class="container-logs">
    <div class="logs-header">
      <select v-model="selectedContainer">
        <option v-for="c in containers" :key="c.name" :value="c.name">
          {{ c.name }} ({{ c.image }})
        </option>
      </select>
      <label><input type="checkbox" v-model="autoScroll"> 自动滚动</label>
      <button @click="clearLogs">清空</button>
    </div>
    <div class="logs-body" ref="logsRef">
      <div v-for="(line, i) in logs" :key="i" class="log-line" :class="line.level">
        <span class="log-time">{{ line.time }}</span>
        <span class="log-text">{{ line.text }}</span>
      </div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, watch, nextTick } from 'vue'
import { getSandboxLogs } from '../../sdk'

const props = defineProps<{ sandboxId: string, containers: any[] }>()

const selectedContainer = ref(props.containers[0]?.name)
const logs = ref([])
const autoScroll = ref(true)
const logsRef = ref(null)

// 轮询日志（Phase 2 改为 WebSocket）
let timer: any
watch(selectedContainer, async (name) => {
  clearInterval(timer)
  logs.value = []
  const fetchLogs = async () => {
    const newLogs = await getSandboxLogs(props.sandboxId, name, logs.value.length)
    logs.value.push(...newLogs)
    if (autoScroll.value) {
      nextTick(() => { logsRef.value?.scrollTo(0, logsRef.value.scrollHeight) })
    }
  }
  await fetchLogs()
  timer = setInterval(fetchLogs, 2000)
}, { immediate: true })
</script>
```

```python
class ProcessDefinition(BaseModel):
    """BPMN 流程定义 — 从 platform 或本地 YAML 加载"""

    name: str
    description: str = ""
    nodes: List[dict]                 # EventNode / ActivityNode / GatewayNode 的 dict
    edges: List[dict]                 # FlowEdge 的 dict

    artifacts: List[dict] = []


class ProcessInstance(BaseModel):
    """流程实例 — 运行时状态"""
    instance_id: str
    process_id: str                   # 来自哪个 ProcessDefinition

    node_outputs: Dict[str, dict] = {} # { node_id: output_dict }

    updated_at: datetime
```

## 三、核心执行流程

### 3.1 daemon 事件循环

```python
# daemon/event_loop.py

class EventLoop:
    """flow daemon 的主事件循环"""

    async def run(self, project_id: str):
        """启动事件循环"""

        # 1. 加载或恢复状态
        self.state = await self.state_persister.load(project_id)
        if self.state is None:
            self.state = await self.initialize_project(project_id)

        # 2. 进入事件循环
        while True:
            event = await self.wait_for_event()

            match event.type:
                case "node_result":
                    await self.handle_node_result(event.data)
                case "gate_response":
                    await self.handle_gate_response(event.data)
                case "user_message":
                    await self.handle_user_message(event.data)
                case "platform_event":
                    await self.handle_platform_event(event.data)
                case "timeout":
                    await self.handle_timeout(event.data)

            # 每次事件处理后持久化状态
            await self.state_persister.save(self.state)


    async def initialize_project(self, project_id: str) -> FlowState:
        """初始化新项目"""

        # 1. 从 platform 加载项目配置
        project = await self.platform_client.get_project(project_id)
        pipeline_id = project["pipeline_process_id"]

        # 2. 加载 Pipeline ProcessDefinition
        pipeline_def = await self.platform_client.get_process(pipeline_id)

        # 3. 实例化 Pipeline
        pipeline_instance = self.process_manager.instantiate(pipeline_def, project_id)

        # 4. 创建 FlowState
        state = FlowState(
            project_id=project_id,
            pipeline_instance=pipeline_instance,
            updated_at=datetime.now(),
        )

        # 5. 推进到第一个阶段
        await self.advance_pipeline(state)

        return state
```

### 3.2 节点调度（NodeDispatcher）

```python
# engine/node_dispatcher.py

class NodeDispatcher:
    """按节点类型分发执行"""

    async def dispatch(self, state: FlowState, instance: ProcessInstance, node: dict):
        """处理一个节点"""

        category = node.get("category")

        match category:
            case "event":
                await self.handle_event_node(state, instance, node)
            case "activity":
                await self.handle_activity_node(state, instance, node)
            case "gateway":
                await self.handle_gateway_node(state, instance, node)


    async def handle_event_node(self, state, instance, node):
        """处理事件节点"""
        position = node.get("event_position")

        if position == "start":
            # Start Event → 直接推进到下一个节点
            instance.node_states[node["id"]] = "completed"
            next_node = self.find_next_node(instance, node["id"])
            if next_node:
                await self.dispatch(state, instance, next_node)

        elif position == "end":
            # End Event → 流程实例完成
            instance.node_states[node["id"]] = "completed"
            await self.on_process_completed(state, instance)


    async def handle_activity_node(self, state, instance, node):
        """处理活动节点 → 构造 FlowMessage 发送给 runtime"""
        task_type = node.get("task_type", "abstract")
        sub_process_type = node.get("sub_process_type")

        instance.node_states[node["id"]] = "running"

        if task_type == "user":
            # User Task → 门禁
            await self.gate_controller.create_gate(state, instance, node)
            return  # 等待用户操作

        if sub_process_type == "ad_hoc":
            # Ad-Hoc Sub-Process → session
            msg = self.build_session_message(state, instance, node)
        elif sub_process_type == "call_activity":
            # Call Activity → 启动子流程
            await self.start_sub_process(state, instance, node)
            return
        else:
            # 普通 Task → 任务分配
            msg = self.build_task_message(state, instance, node)

        # 发送 FlowMessage 给 runtime
        await self.runtime_client.send(msg)
        state.pending_messages[msg.message_id] = "dispatched"


    async def handle_gateway_node(self, state, instance, node):
        """处理网关节点 → 条件路由"""
        gateway_type = node.get("gateway_type")

        if gateway_type == "exclusive":
            # 排他网关：读取上一个节点的 output，按条件选择分支
            next_node_id = self.gateway_evaluator.evaluate_exclusive(
                instance, node
            )
            instance.node_states[node["id"]] = "completed"
            next_node = self.get_node(instance, next_node_id)
            await self.dispatch(state, instance, next_node)

        elif gateway_type == "parallel":
            direction = node.get("direction")
            if direction == "diverging":
                # 并行分支：同时启动所有出口分支
                branches = self.find_outgoing_nodes(instance, node["id"])
                instance.node_states[node["id"]] = "completed"
                for branch_node in branches:
                    await self.dispatch(state, instance, branch_node)
            elif direction == "converging":
                # 并行合并：检查所有入口分支是否完成
                if self.all_incoming_completed(instance, node["id"]):
                    instance.node_states[node["id"]] = "completed"
                    next_node = self.find_next_node(instance, node["id"])
                    if next_node:
                        await self.dispatch(state, instance, next_node)
                # 否则等待（其他分支完成时会再次检查）


    def build_task_message(self, state, instance, node) -> FlowMessage:
        """构造任务分配消息"""
        actor = node.get("actor")
        agent_id = self.resolve_actor(state, actor)

        return FlowMessage(
            message_id=generate_id(),
            type="task_assign",
            agent_id=agent_id,
            project_id=state.project_id,
            payload={
                "process_instance_id": instance.instance_id,
                "node_id": node["id"],
                "action": node.get("action", ""),
                "control_level": "enforced" if node.get("call_process_id") else "free",
                "process_id": node.get("call_process_id"),
                "input_refs": node.get("input_refs", []),
                "allowed_tools": node.get("allowed_tools", []),
                "output_schema": node.get("output_schema"),
                "suspicion_hints": self.get_suspicion_hints(node),
            },
            created_at=datetime.now(),
        )


    def build_session_message(self, state, instance, node) -> FlowMessage:
        """构造协作会议消息"""
        session = node.get("session", {})
        moderator_id = self.resolve_actor(state, session.get("moderator"))
        participant_ids = [self.resolve_actor(state, p) for p in session.get("participants", [])]

        return FlowMessage(
            message_id=generate_id(),
            type="session_start",
            agent_id=moderator_id,
            project_id=state.project_id,
            payload={
                "process_instance_id": instance.instance_id,
                "node_id": node["id"],
                "moderator_id": moderator_id,
                "participant_ids": participant_ids,
                "agenda": session.get("agenda", ""),
                "allowed_tools": node.get("allowed_tools", []),
                "output_schema": node.get("output_schema"),
            },
            created_at=datetime.now(),
        )


    async def on_process_completed(self, state: FlowState, instance: ProcessInstance):
        """流程实例完成后的处理"""

        if instance.parent_instance_id is None:
            # 这是 Pipeline 流程完成 → 项目完成
            await self.notify_chat(state, "🎉 项目已完成！")
            return

        # 这是阶段流程完成 → 推进 Pipeline 到下一个阶段
        pipeline = state.pipeline_instance

        # 找到当前阶段在 Pipeline 中的节点
        current_stage_node_id = None
        for node_id, inst_id in state.stage_instance_map.items():
            if inst_id == instance.instance_id:
                current_stage_node_id = node_id
                break

        if current_stage_node_id:
            pipeline.node_states[current_stage_node_id] = "completed"
            pipeline.node_outputs[current_stage_node_id] = {
                "completed_at": datetime.now().isoformat(),
            }

            # 记录 edge_history
            next_edge = self.find_outgoing_edge(pipeline, current_stage_node_id)
            if next_edge:
                pipeline.edge_history.append(EdgeTraversal(
                    edge_id=next_edge["id"],
                    source_node_id=current_stage_node_id,
                    target_node_id=next_edge["target"],
                    traversed_at=datetime.now(),
                ))

            # 推进到下一个节点（可能是 Gateway 门禁或下一个 Call Activity）
            next_node = self.find_next_node(pipeline, current_stage_node_id)
            if next_node:
                await self.dispatch(state, pipeline, next_node)
            else:
                # Pipeline 没有下一个节点 → 项目完成
                await self.on_process_completed(state, pipeline)

            # 通知
            await self.notify_chat(state,
                f"🔔 阶段完成：{current_stage_node_id}",
                link_type="stage", link_id=current_stage_node_id)

        # 持久化
        await self.state_persister.save(state)


    async def start_sub_process(self, state: FlowState, parent_instance: ProcessInstance, node: dict):
        """启动子流程（Call Activity）"""
        call_process_id = node.get("call_process_id")

        # 从 platform 加载子流程定义
        sub_def = await self.platform_client.get_process(call_process_id)

        # 实例化子流程
        sub_instance = self.process_manager.instantiate(sub_def, state.project_id)
        sub_instance.parent_instance_id = parent_instance.instance_id

        # 记录映射关系（阶段节点 ID → 子流程实例 ID）
        if not hasattr(state, 'stage_instance_map'):
            state.stage_instance_map = {}
        state.stage_instance_map[node["id"]] = sub_instance.instance_id

        # 如果是阶段流程，设为当前 stage_instance
        if parent_instance.instance_id == state.pipeline_instance.instance_id:
            state.stage_instance = sub_instance

        # 推进子流程的第一个节点
        start_node = self.find_start_node(sub_instance)
        await self.dispatch(state, sub_instance, start_node)

        # 持久化
        await self.state_persister.save(state)


    def get_suspicion_hints(self, node: dict) -> List[str]:
        """根据节点类型生成怀疑机制提示"""
        hints = []
        node_name = node.get("name", "")
        node_id = node.get("id", "")

        # 代码审查节点
        if "review" in node_id or "review" in node_name.lower():
            hints.append("前序产出（PRD、设计文档）可能有遗漏或错误，请独立验证代码是否满足实际需求。")

        # 测试节点
        if "test" in node_id or "test" in node_name.lower():
            hints.append("实现代码可能有 BUG，PRD 可能有遗漏，请从用户视角独立设计测试用例。")

        # 修复节点（审查打回后）
        if "fix" in node_id:
            hints.append("前序产出可能有其他遗漏，请独立验证，不要只修复指出的问题。")

        return hints


    async def advance_to_next(self, state, instance, current_node_id, next_node_id, edge):
        """推进到下一个节点，记录 edge_history + 持久化"""

        # 记录 edge_history
        existing = next((e for e in instance.edge_history if e.edge_id == edge["id"]), None)
        if existing:
            existing.count += 1
            existing.traversed_at = datetime.now()
        else:
            instance.edge_history.append(EdgeTraversal(
                edge_id=edge["id"],
                source_node_id=current_node_id,
                target_node_id=next_node_id,
                label=edge.get("label"),
                traversed_at=datetime.now(),
            ))

        # 持久化（关键步骤）
        await self.state_persister.save(state)

        # 推进
        instance.current_node_id = next_node_id
        next_node = self.get_node(instance, next_node_id)
        await self.dispatch(state, instance, next_node)
```

```python
class EventLoop:
    async def handle_node_result(self, result: NodeResult):
        """处理 runtime 上报的执行结果"""

        # 1. 更新 pending_messages
        self.state.pending_messages[result.message_id] = result.result_type

        # 2. 找到对应的 ProcessInstance 和节点
        instance, node = self.find_instance_and_node(result.message_id)

        # 3. 判断是否是 session 节点（moderator 提交结论 = session 结束）
        if node.get("sub_process_type") == "ad_hoc":
            # session 结束 → 通知 runtime 结束会议
            await self.runtime_client.send(FlowMessage(
                message_id=generate_id(),
                type="session_end",
                agent_id=node.get("session", {}).get("moderator", ""),
                project_id=self.state.project_id,
                payload={"process_instance_id": instance.instance_id, "node_id": node["id"]},
                created_at=datetime.now(),
            ))

        # 4. 根据结果类型处理
        match result.result_type:
            case "completed":
                instance.node_states[node["id"]] = "completed"
                instance.node_outputs[node["id"]] = result.output or {}

                # 通知
                await self.notify_chat(self.state,
                    f"✅ {node.get('name', node['id'])} 已完成",
                    link_type="task", link_id=node["id"])

                # 检查是否是并行分支中的节点 → 检查 join 网关
                join_node = self.find_converging_gateway(instance, node["id"])
                if join_node:
                    # 尝试推进 join 网关（如果所有分支都完成了）
                    await self.node_dispatcher.handle_gateway_node(self.state, instance, join_node)
                else:
                    # 推进到下一个节点（通过 advance_to_next 记录 edge_history）
                    next_edge = self.find_outgoing_edge(instance, node["id"])
                    if next_edge:
                        await self.node_dispatcher.advance_to_next(
                            self.state, instance, node["id"], next_edge["target"], next_edge)

                # 如果是 TaskInstance 的工作流完成 → 更新任务状态
                task = self.find_task_by_workflow(instance)
                if task and self.is_workflow_completed(instance):
                    await self.task_scheduler.on_task_completed(self.state, task.instance_id)

            case "blocked":
                instance.node_states[node["id"]] = "blocked"
                await self.notify_chat(self.state,
                    f"⚠️ {result.agent_id} 被阻塞：{result.reason}",
                    link_type="task", link_id=node["id"])

            case "failed":
                instance.node_states[node["id"]] = "failed"

                # 检查重试次数
                retry_count = instance.node_outputs.get(f"_retry_{node['id']}", 0)
                max_retries = node.get("max_retries", 2)

                if retry_count < max_retries:
                    # 重试：重新执行同一个节点
                    instance.node_outputs[f"_retry_{node['id']}"] = retry_count + 1
                    instance.node_states[node["id"]] = "pending"
                    await self.notify_chat(self.state,
                        f"🔄 {node.get('name', node['id'])} 失败，重试 ({retry_count + 1}/{max_retries})")
                    await self.node_dispatcher.dispatch(self.state, instance, node)
                else:
                    # 超过重试次数 → 检查 on_fail 边
                    fail_edge = self.find_fail_edge(instance, node["id"])
                    if fail_edge:
                        await self.notify_chat(self.state,
                            f"↩ {node.get('name', node['id'])} 失败，回退到 {fail_edge['target']}")
                        await self.node_dispatcher.advance_to_next(
                            self.state, instance, node["id"], fail_edge["target"], fail_edge)
                    else:
                        # 没有 on_fail 边 → 流程卡住，通知用户
                        await self.notify_chat(self.state,
                            f"❌ {result.agent_id} 执行失败且无回退路径：{result.reason}")

        # 5. 持久化
        await self.state_persister.save(self.state)
```

```python
# engine/gateway_evaluator.py

class GatewayEvaluator:
    """条件表达式求值"""

    def evaluate_exclusive(self, instance: ProcessInstance, gateway_node: dict) -> str:
        """排他网关：返回应该走的下一个节点 ID"""

        # 找到所有出口边
        outgoing_edges = self.find_outgoing_edges(instance, gateway_node["id"])

        # 获取上一个节点的 output
        prev_node_id = self.find_previous_node(instance, gateway_node["id"])
        prev_output = instance.node_outputs.get(prev_node_id, {})

        # 逐个检查条件
        for edge in outgoing_edges:
            condition = edge.get("condition")
            if condition and self.evaluate_condition(condition, prev_output):
                return edge["target"]

        # 没有条件匹配 → 走 default 边
        for edge in outgoing_edges:
            if edge.get("flow_type") == "default":
                return edge["target"]

        raise NoMatchingCondition(f"Gateway {gateway_node['id']} no matching condition")


    def evaluate_condition(self, condition: str, context: dict) -> bool:
        """求值条件表达式
        Phase 1：简单字符串匹配
        Phase 2：表达式引擎
        """
        # 例：condition = "verdict == 'approved'"
        # 解析为：context["verdict"] == "approved"
        parts = condition.split("==")
        if len(parts) == 2:
            field = parts[0].strip()
            value = parts[1].strip().strip("'\"")
            return str(context.get(field)) == value

        parts = condition.split("!=")
        if len(parts) == 2:
            field = parts[0].strip()
            value = parts[1].strip().strip("'\"")
            return str(context.get(field)) != value

        return False
```

```python
            # 推进到下一个节点
            instance.node_states[node["id"]] = "completed"

                await self.node_dispatcher.dispatch(state, instance, next_node)

        await self.send_gate_card(state, gate)


    async def handle_gate_response(self, state: FlowState, gate_id: str, action: str, comment: str = ""):
        """处理用户的审批操作"""
        gate = self.find_pending_gate(state, gate_id)
        if not gate:
            return

        gate.responded_at = datetime.now()
        gate.comment = comment

        match action:
            case "approve":
                gate.status = "approved"
                state.gate_history.append(gate.dict())
                state.pending_gates.remove(gate)
                # 推进流程
                instance, node = self.find_gate_context(state, gate)
                instance.node_states[node["id"]] = "completed"
                next_node = self.find_next_node(instance, node["id"])
                if next_node:
                    await self.node_dispatcher.dispatch(state, instance, next_node)

            case "request_changes":
                gate.status = "changes_requested"
                state.pending_gates.remove(gate)
                # 回退到指定节点
                instance, node = self.find_gate_context(state, gate)
                rollback_target = node.get("on_request_changes") or self.find_previous_activity(instance, node["id"])
                if rollback_target:
                    # 注入用户意见到下一次任务的 action
                    await self.rollback_with_feedback(state, instance, rollback_target, comment)

            case "reject":
                gate.status = "rejected"
                state.gate_history.append(gate.dict())
                state.pending_gates.remove(gate)
                # 回退到更早的阶段或终止
                await self.handle_rejection(state, gate)
```

```python
# business/task_scheduler.py

class TaskScheduler:
    """阶段 3 的任务调度器"""

    async def initialize(self, state: FlowState, workitem_ids: List[str]):
        """从平台工单列表初始化 TaskInstance + 依赖图"""

        for wid in workitem_ids:
            workitem = await self.platform_client.get_workitem(wid)
            task = TaskInstance(
                instance_id=f"taski-{wid}",
                source_ref=wid,
                task_type=workitem.get("item_type", "task"),
                status="draft",
                owner_id=workitem.get("assignee_id", ""),
                reviewer_id="tech-lead",
                dependencies=[f"taski-{d}" for d in workitem.get("dependencies", [])],
                wave=self.calculate_wave(workitem),
                priority=workitem.get("priority", "medium"),
            )
            state.task_instances.append(task)

        # 构建依赖图
        state.dependency_graph = self.build_dependency_graph(state.task_instances)

        # 标记无依赖的任务为 ready
        for task in state.task_instances:
            if not task.dependencies:
                task.status = "ready"

        # 调度就绪任务
        await self.dispatch_ready_tasks(state)


    async def dispatch_ready_tasks(self, state: FlowState):
        """调度所有就绪的任务"""
        ready = [t for t in state.task_instances if t.status == "ready"]

        for task in ready:
            # 确保 Agent 已创建
            if task.owner_id not in state.active_agents:
                await self.agent_lifecycle.create_agent(state, task.owner_id)

            # 为任务创建 ProcessInstance（iterative_development 工作流）
            workflow_def = await self.platform_client.get_process("iterative_development")
            task_instance = self.process_manager.instantiate(workflow_def, state.project_id)
            task_instance.parent_instance_id = state.stage_instance.instance_id
            task.workflow_instance_id = task_instance.instance_id
            task.status = "in_progress"

            # 更新平台工单状态
            await self.platform_client.transition_workitem(task.source_ref, "in_progress")

            # 推进工作流的第一个节点
            start_node = self.find_start_node(task_instance)
            await self.node_dispatcher.dispatch(state, task_instance, start_node)


    async def on_task_completed(self, state: FlowState, task_id: str):
        """任务完成后，检查下游任务是否就绪"""
        task = self.find_task(state, task_id)
        task.status = "completed"

        # 更新平台工单状态
        await self.platform_client.transition_workitem(task.source_ref, "completed")

        # 检查下游任务
        newly_ready = self.dependency_graph.on_task_completed(task_id)
        for t in newly_ready:
            t.status = "ready"

        # 调度新就绪的任务
        await self.dispatch_ready_tasks(state)

        # 检查是否所有任务完成
        if all(t.status == "completed" for t in state.task_instances):
            # 阶段 3 完成 → 推进到集成测试或下一阶段
            await self.on_all_tasks_completed(state)
```

```python
    if link_type and link_id:
        content += f" → 查看 [{link_type}:{link_id}]"
    
    # 通过 gate 推送给 chat 前端
    await self.gate_client.push_event({
        "type": "Notification",
        "project_id": state.project_id,
        "data": {
            "text": content,
            "link_type": link_type,   # "task" | "bug" | "stage" | "gate"
            "link_id": link_id,       # "TASK-003" | "BUG-002" | "requirement"
        }
    })
```

### 4.6 edge_history 数据结构

记录哪些边被走过，用于前端标注走过的路径和打回路径。

```python
class EdgeTraversal(BaseModel):
    """边的遍历记录"""
    edge_id: str
    source_node_id: str
    target_node_id: str
    label: Optional[str] = None       # 条件标签（如 "approved"、"needs_revision"）
    traversed_at: datetime
    count: int = 1                    # 这条边被走过几次（打回会 >1）
```

在 ProcessInstance 中新增：

```python
class ProcessInstance(BaseModel):
    # ... 原有字段 ...
    edge_history: List[EdgeTraversal] = []  # 新增：边遍历历史
```

NodeDispatcher 每次推进节点时记录：

```python
async def advance_to_next(self, instance, current_node_id, next_node_id, edge):
    """推进到下一个节点，记录边遍历"""
    # 记录 edge_history
    existing = next((e for e in instance.edge_history if e.edge_id == edge["id"]), None)
    if existing:
        existing.count += 1
        existing.traversed_at = datetime.now()
    else:
        instance.edge_history.append(EdgeTraversal(
            edge_id=edge["id"],
            source_node_id=current_node_id,
            target_node_id=next_node_id,
            label=edge.get("label"),
            traversed_at=datetime.now(),
        ))
    
    # 推进
    instance.current_node_id = next_node_id
    await self.dispatch(state, instance, self.get_node(instance, next_node_id))
```

## 六、Phase 分期

### Phase 1

| 模块 | 范围 |
|------|------|
| daemon/event_loop | 基础事件循环（NodeResult + gate_response） |
| engine/process_manager | ProcessDefinition 加载、实例化 |
| engine/node_dispatcher | Activity → FlowMessage、Gateway → 条件路由、Event → start/end |
| engine/gateway_evaluator | 简单字符串匹配（== / !=） |
| business/gate_controller | hard/soft 门禁、审批卡片 |
| business/task_scheduler | 依赖图、波次调度、并行任务 |
| business/agent_lifecycle | agent_create / agent_destroy |
| infra/state_persister | 文件持久化（current.yaml） |
| infra/runtime_client | FlowMessage 发送、NodeResult 接收 |
| infra/platform_client | workitem / process_template API |

Phase 1 不实现：事件日志、agent_suspend/resume、表达式引擎

> **设计决策：Parallel Gateway 和 Call Activity 在 Phase 1 实现**
>
> 原计划 Phase 2 才实现，但原型验证发现：
> - 阶段 3 并行任务需要 Parallel Gateway（fork/join）
> - Pipeline 的每个阶段就是 Call Activity（启动子流程）
> - 没有这两个能力，Phase 1 无法跑通完整的六阶段流程
>
> 因此调整为 Phase 1 实现基础版（不含嵌套并行、不含动态分支数量）

## 二、认证授权

### 2.1 JWT 签发与校验

```rust
// auth/jwt.rs
//
// JWT 认证服务 — 签发和校验 JSON Web Token。
//
// 设计意图：
//   gate 是所有客户端请求的唯一入口，负责认证后将用户身份注入请求头，
//   后端服务（platform/flow/runtime）信任 gate 注入的 X-User-Id / X-User-Role。
//
// Phase 1：简单 JWT（HS256 对称加密，secret 从配置文件读取）
// Phase 2：RS256 非对称加密 + refresh token + token 黑名单

use jsonwebtoken::{encode, decode, Header, Validation, EncodingKey, DecodingKey};

/// JWT Claims — token 中携带的用户信息
#[derive(Debug, Serialize, Deserialize)]
pub struct Claims {
    pub sub: String,          // 用户标识（user_id）
    pub role: String,         // 用户角色：admin | user（决定 API 访问权限）
    pub exp: usize,           // 过期时间（Unix timestamp，秒）
    pub iat: usize,           // 签发时间（Unix timestamp）
}

/// JWT 服务 — 封装签发和校验逻辑
///
/// 为什么独立为 struct 而非全局函数：
///   1. 持有 key 和配置（避免每次调用都读配置）
///   2. 可以注入到 axum State 中（Arc<JwtService>）
///   3. 测试时可以构造不同配置的实例
pub struct JwtService {
    encoding_key: EncodingKey,   // 签发用的密钥
    decoding_key: DecodingKey,   // 校验用的密钥（HS256 时两者相同）
    expiry_hours: u64,           // token 有效期（小时）— 默认 24h
}

impl JwtService {
    pub fn new(secret: &str, expiry_hours: u64) -> Self {
        Self {
            encoding_key: EncodingKey::from_secret(secret.as_bytes()),
            decoding_key: DecodingKey::from_secret(secret.as_bytes()),
            expiry_hours,
        }
    }

    pub fn sign(&self, user_id: &str, role: &str) -> Result<String, GateError> {
        let now = chrono::Utc::now().timestamp() as usize;
        let claims = Claims {
            sub: user_id.to_string(),
            role: role.to_string(),
            exp: now + (self.expiry_hours * 3600) as usize,
            iat: now,
        };
        encode(&Header::default(), &claims, &self.encoding_key)
            .map_err(|e| GateError::JwtError(e.to_string()))
    }

    pub fn verify(&self, token: &str) -> Result<Claims, GateError> {
        decode::<Claims>(token, &self.decoding_key, &Validation::default())
            .map(|data| data.claims)
            .map_err(|e| GateError::Unauthorized(e.to_string()))
    }
}
```

### 2.2 认证中间件

```rust
// auth/middleware.rs
//
// 认证中间件 — 拦截所有请求，校验 JWT，注入用户身份到请求头。
//
// 白名单路径（不需要认证）：
//   /auth/* — 登录/注册/刷新 token
//   /health — 健康检查（k8s probe 用）
//
// 认证成功后注入的请求头：
//   X-User-Id — 用户标识（后端服务据此判断"谁在操作"）
//   X-User-Role — 用户角色（后端服务据此判断"有没有权限"）
//   后端服务信任这些头（因为只有 gate 能注入，客户端无法伪造）

pub async fn auth_middleware(
    State(jwt): State<Arc<JwtService>>,
    mut req: Request,
    next: Next,
) -> Result<Response, GateError> {
    // 跳过不需要认证的路径
    let path = req.uri().path();
    if path.starts_with("/auth/") || path == "/health" {
        return Ok(next.run(req).await);
    }

    // 从 Authorization header 提取 token
    let token = req.headers()
        .get("Authorization")
        .and_then(|v| v.to_str().ok())
        .and_then(|v| v.strip_prefix("Bearer "))
        .ok_or(GateError::Unauthorized("Missing Authorization header".into()))?;

    // 校验 JWT
    let claims = jwt.verify(token)?;

    // 注入用户信息到请求头（后端服务可读取）
    req.headers_mut().insert("X-User-Id", claims.sub.parse().unwrap());
    req.headers_mut().insert("X-User-Role", claims.role.parse().unwrap());

    Ok(next.run(req).await)
}
```

### 2.3 登录 Handler

```rust
// auth/handler.rs
//
// 登录 handler — Phase 1 简单实现（配置文件用户列表）。
// Phase 2 改为调用 platform 的用户认证 API。

#[derive(Deserialize)]
pub struct LoginRequest {
    pub email: String,            // 用户邮箱
    pub password: String,         // 密码（Phase 1 明文比对，Phase 2 bcrypt）
}

#[derive(Serialize)]
pub struct LoginResponse {
    pub token: String,            // JWT token（客户端存储后每次请求携带）
    pub user_id: String,          // 用户 ID
    pub expires_at: String,       // 过期时间（ISO 8601，前端据此判断是否需要刷新）
}

pub async fn login(
    State(state): State<AppState>,
    Json(req): Json<LoginRequest>,
) -> Result<Json<LoginResponse>, GateError> {
    // Phase 1：简单验证（从配置文件读取用户列表）
    // Phase 2：调用 platform 的用户认证 API
    let user = state.config.find_user(&req.email, &req.password)
        .ok_or(GateError::Unauthorized("Invalid credentials".into()))?;

    let token = state.jwt.sign(&user.id, &user.role)?;

    Ok(Json(LoginResponse {
        token,
        user_id: user.id.clone(),
        expires_at: (chrono::Utc::now() + chrono::Duration::hours(state.config.auth.token_expiry_hours as i64)).to_rfc3339(),
    }))
}
```

### 3.2 反向代理实现

```rust
// router/proxy.rs
//
// HTTP 反向代理 — 将客户端请求转发给后端服务。
//
// 设计意图：
//   客户端只需要知道 gate 的地址，不需要知道后端服务的内部地址。
//   gate 负责：路由匹配 → 复制 headers → 复制 body → 转发 → 返回响应。
//
// body 大小限制（10MB）：
//   防止客户端发送超大请求导致 OOM。
//   10MB 对于 JSON API 足够（最大的请求是产物上传，Phase 2 改为流式传输）。
//
// 错误处理：
//   路由不匹配 → 404 NotFound
//   后端服务不可达 → 502 UpstreamError
//   body 超大 → 400 BadRequest

pub async fn proxy(
    State(state): State<AppState>,
    req: Request,
) -> Result<Response, GateError> {
    let path = req.uri().path();

    // 查找匹配的路由
    let route = state.config.routes.iter()
        .find(|r| path.starts_with(&r.prefix))
        .ok_or(GateError::NotFound(format!("No route for {}", path)))?;

    let target_url = format!("{}{}", route.target, path);

    // 转发请求
    let mut proxy_req = state.http_client
        .request(req.method().clone(), &target_url);

    // 复制 headers（保留 X-User-Id / X-Request-Id）
    for (key, value) in req.headers() {
        if key != "host" {
            proxy_req = proxy_req.header(key, value);
        }
    }

    // 复制 body
    let body = axum::body::to_bytes(req.into_body(), 10 * 1024 * 1024).await
        .map_err(|_| GateError::BadRequest("Body too large".into()))?;
    proxy_req = proxy_req.body(body);

    // 发送并返回响应
    let resp = proxy_req.send().await
        .map_err(|e| GateError::UpstreamError(e.to_string()))?;

    let status = resp.status();
    let headers = resp.headers().clone();
    let body = resp.bytes().await
        .map_err(|e| GateError::UpstreamError(e.to_string()))?;

    let mut response = Response::new(body.into());
    *response.status_mut() = status;
    *response.headers_mut() = headers;
    Ok(response)
}
```

### 3.3 请求 ID 中间件

```rust
// router/middleware.rs
//
// 请求 ID 中间件 — 为每个请求生成唯一标识，贯穿整条调用链。
//
// 设计意图：
//   X-Request-Id 是分布式追踪的基础。gate 生成后注入请求头，
//   后端服务（platform/flow/runtime）继承此 ID 记录到日志和事件中。
//   出问题时按 request_id 过滤即可看到完整调用链。

pub async fn request_id_middleware(
    mut req: Request,
    next: Next,
) -> Response {
    let request_id = uuid::Uuid::new_v4().to_string();
    req.headers_mut().insert("X-Request-Id", request_id.parse().unwrap());
    let mut resp = next.run(req).await;
    resp.headers_mut().insert("X-Request-Id", request_id.parse().unwrap());
    resp
}
```

## 四、WebSocket 管理

### 4.1 连接升级

```rust
// ws/handler.rs
//
// WebSocket 连接管理 — 客户端通过 WS 接收实时事件（Agent 执行过程、聊天消息、状态变更）。
//
// 连接生命周期：
//   1. 客户端发起 WS 升级请求（携带 JWT token）
//   2. gate 校验 token → 升级为 WebSocket 连接
//   3. 注册 session → 客户端发送 subscribe 消息订阅 topic
//   4. 后端事件到达 → gate 按 topic 广播给订阅者
//   5. 客户端断开 → 清理 session 和订阅
//
// channel 容量（256）：
//   每个 session 的消息缓冲区。如果客户端消费太慢（网络差），
//   超过 256 条未发送的消息会导致 channel 满，新消息被丢弃。
//   256 对于正常网络足够（峰值约 10 条/秒）。

pub async fn ws_upgrade(
    State(state): State<AppState>,
    ws: WebSocketUpgrade,
    headers: HeaderMap,
) -> Result<Response, GateError> {
    // 从 query param 或 header 获取 token
    let token = headers.get("Authorization")
        .and_then(|v| v.to_str().ok())
        .and_then(|v| v.strip_prefix("Bearer "))
        .ok_or(GateError::Unauthorized("WS requires token".into()))?;

    let claims = state.jwt.verify(token)?;

    Ok(ws.on_upgrade(move |socket| handle_ws(socket, claims, state)))
}

async fn handle_ws(socket: WebSocket, claims: Claims, state: AppState) {
    let (tx, mut rx) = socket.split();
    let (msg_tx, msg_rx) = mpsc::channel(256);

    // 注册 session
    let session = WsSession {
        user_id: claims.sub.clone(),
        subscriptions: HashSet::new(),
        tx: msg_tx,
    };
    let session_id = state.ws_manager.register(session).await;

    // 发送任务：msg_rx → WebSocket tx
    let send_task = tokio::spawn(async move {
        let mut msg_rx = msg_rx;
        let mut tx = tx;
        while let Some(msg) = msg_rx.recv().await {
            let text = serde_json::to_string(&msg).unwrap();
            if tx.send(Message::Text(text)).await.is_err() {
                break;
            }
        }
    });

    // 接收任务：WebSocket rx → 处理
    while let Some(Ok(msg)) = rx.next().await {
        match msg {
            Message::Text(text) => {
                if let Ok(ws_msg) = serde_json::from_str::<ClientWsMessage>(&text) {
                    state.ws_manager.handle_client_message(session_id, ws_msg).await;
                }
            }
            Message::Ping(data) => {
                let _ = state.ws_manager.send(session_id, WsMessage::Pong(data)).await;
            }
            Message::Close(_) => break,
            _ => {}
        }
    }

    // 清理
    state.ws_manager.unregister(session_id).await;
    send_task.abort();
}
```

### 4.2 订阅管理

```rust
// ws/subscription.rs
//
// WebSocket 订阅管理 — 客户端通过 subscribe/unsubscribe 消息管理感兴趣的 topic。
//
// topic 命名规则：
//   "project:{project_id}" — 项目级事件（工单变更、阶段推进、Agent 状态）
//   "chat:{project_id}" — 项目群聊消息（Agent 之间的对话）
//   "chat:assistant:{user_id}" — Assistant 私聊（用户与 Assistant 的 1:1 对话）
//
// ClientWsMessage 三种类型：
//   Subscribe — 订阅 topic（连接后第一件事）
//   Unsubscribe — 取消订阅（切换项目时）
//   Chat — 用户发送聊天消息（gate 转发给 flow，flow 路由给 Agent）

#[derive(Debug, Deserialize)]
#[serde(tag = "type")]
pub enum ClientWsMessage {
    #[serde(rename = "subscribe")]
    Subscribe { topic: String },
    #[serde(rename = "unsubscribe")]
    Unsubscribe { topic: String },
    #[serde(rename = "chat")]
    Chat { room_id: String, content: String },
}

// topic 格式：
//   "project:{project_id}"          — 项目级事件（工单变更、阶段推进）
//   "chat:{project_id}"             — 项目群聊消息
//   "chat:assistant:{user_id}"      — Assistant 私聊

impl WsManager {
    pub async fn handle_client_message(&self, session_id: usize, msg: ClientWsMessage) {
        match msg {
            ClientWsMessage::Subscribe { topic } => {
                self.subscribe(session_id, &topic).await;
            }
            ClientWsMessage::Unsubscribe { topic } => {
                self.unsubscribe(session_id, &topic).await;
            }
            ClientWsMessage::Chat { room_id, content } => {
                // 转发给 flow（flow 负责消息路由）
                self.forward_chat(&room_id, &content, session_id).await;
            }
        }
    }
}
```

```rust
impl WsManager {
```

## 五、限流

```rust
// ratelimit/token_bucket.rs
//
// 限流服务 — 基于 Token Bucket 算法，按用户 ID 限制请求频率。
//
// 设计意图：
//   防止单个用户（或失控的 Agent）压垮后端服务。
//   每个用户独立的限流器，互不影响。
//
// 算法：Token Bucket（令牌桶）
//   - 桶容量 = burst_size（允许短时间突发）
//   - 填充速率 = requests_per_second（稳态速率）
//   - 请求到达时取一个 token，桶空则拒绝（429）
//
// 为什么用 DashMap 而非 HashMap + Mutex：
//   DashMap 是分段锁（sharded lock），高并发下性能远优于全局 Mutex。
//   每个用户的 limiter 独立，不同用户的请求不会互相阻塞。
//
// ws_connections_per_user：
//   限制单个用户的 WebSocket 连接数（防止恶意开大量连接）。
//   正常使用只需 1–2 个连接（一个 chat + 一个 console）。

use governor::{Quota, RateLimiter};
use std::num::NonZeroU32;

pub struct RateLimitService {
    limiters: DashMap<String, RateLimiter<...>>,  // 按 user_id 分桶
    config: RateLimitConfig,
}

#[derive(Debug, Deserialize)]
pub struct RateLimitConfig {
    pub requests_per_second: u32,        // 稳态速率（如 50 req/s）
    pub burst_size: u32,                 // 突发容量（如 100，允许短时间 100 个请求）
    pub ws_connections_per_user: u32,    // 单用户最大 WS 连接数（如 5）
}

impl RateLimitService {
    pub fn check(&self, user_id: &str) -> Result<(), GateError> {
        let limiter = self.limiters
            .entry(user_id.to_string())
            .or_insert_with(|| {
                RateLimiter::direct(Quota::per_second(
                    NonZeroU32::new(self.config.requests_per_second).unwrap()
                ).allow_burst(
                    NonZeroU32::new(self.config.burst_size).unwrap()
                ))
            });

        limiter.check()
            .map_err(|_| GateError::RateLimited("Too many requests".into()))
    }
}
```

## 六、错误处理

```rust
// src/error.rs
//
// 统一错误类型 — gate 所有 handler 和中间件的错误返回格式。
//
// 设计意图：
//   与 platform 的 PlatformError 类似，统一 JSON 格式返回。
//   gate 特有的错误类型：UpstreamError（后端不可达）和 RateLimited（限流）。
//
// 错误分类：
//   客户端错误（4xx）：Unauthorized / NotFound / BadRequest / RateLimited
//   服务端错误（5xx）：UpstreamError（后端服务不可达 → 502 Bad Gateway）
//
// UpstreamError 的触发场景：
//   - platform/flow/runtime 服务未启动
//   - 网络超时（默认 30s）
//   - 后端返回非 HTTP 响应（连接被重置等）

#[derive(Debug, thiserror::Error)]
pub enum GateError {
    #[error("Unauthorized: {0}")]
    Unauthorized(String),        // 认证失败 — token 缺失/过期/无效

    #[error("JWT error: {0}")]
    JwtError(String),            // JWT 签发/校验内部错误（密钥问题等）

    #[error("Not found: {0}")]
    NotFound(String),            // 路由不匹配 — 没有对应的后端服务

    #[error("Bad request: {0}")]
    BadRequest(String),          // 请求格式错误 — body 超大 / 缺少必要 header

    #[error("Upstream error: {0}")]
    UpstreamError(String),       // 后端服务不可达 — 502 Bad Gateway

    #[error("Rate limited: {0}")]
    RateLimited(String),         // 限流触发 — 429 Too Many Requests
}

impl IntoResponse for GateError {
    fn into_response(self) -> Response {
        let (status, code) = match &self {
            Self::Unauthorized(_) => (StatusCode::UNAUTHORIZED, "UNAUTHORIZED"),
            Self::JwtError(_) => (StatusCode::UNAUTHORIZED, "JWT_ERROR"),
            Self::NotFound(_) => (StatusCode::NOT_FOUND, "NOT_FOUND"),
            Self::BadRequest(_) => (StatusCode::BAD_REQUEST, "BAD_REQUEST"),
            Self::UpstreamError(_) => (StatusCode::BAD_GATEWAY, "UPSTREAM_ERROR"),
            Self::RateLimited(_) => (StatusCode::TOO_MANY_REQUESTS, "RATE_LIMITED"),
        };

        let body = serde_json::json!({
            "error": { "code": code, "message": self.to_string() }
        });

        (status, axum::Json(body)).into_response()
    }
}
```

|------|------|

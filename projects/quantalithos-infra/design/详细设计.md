## 二、Docker Compose

```yaml
# docker/docker-compose.yml

version: "3.8"

services:
  # ── 基础设施 ──
  postgres:
    image: postgres:16
    environment:
      POSTGRES_DB: quantalithos
      POSTGRES_USER: quantalithos
      POSTGRES_PASSWORD: ${DB_PASSWORD:-dev_password}
    ports:
      - "5432:5432"
    volumes:
      - pgdata:/var/lib/postgresql/data

  redis:
    image: redis:7-alpine
    ports:
      - "6379:6379"

  minio:
    image: minio/minio
    command: server /data --console-address ":9001"
    environment:
      MINIO_ROOT_USER: ${MINIO_USER:-minioadmin}
      MINIO_ROOT_PASSWORD: ${MINIO_PASSWORD:-minioadmin}
    ports:
      - "9000:9000"
      - "9001:9001"
    volumes:
      - miniodata:/data

  # ── 应用服务 ──
  platform:
    build:
      context: ..
      dockerfile: docker/Dockerfile.platform
    environment:
      DATABASE_URL: postgres://quantalithos:${DB_PASSWORD:-dev_password}@postgres:5432/quantalithos
      REDIS_URL: redis://redis:6379
    ports:
      - "8001:8001"
    depends_on:
      - postgres
      - redis

  flow:
    build:
      context: ..
      dockerfile: docker/Dockerfile.flow
    environment:
      PLATFORM_URL: http://platform:8001
      RUNTIME_URL: http://runtime:8003
    ports:
      - "8002:8002"
    depends_on:
      - platform

  runtime:
    build:
      context: ..
      dockerfile: docker/Dockerfile.runtime
    environment:
      PLATFORM_URL: http://platform:8001
      LLM_API_KEY: ${LLM_API_KEY}
      LLM_MODEL: ${LLM_MODEL:-claude-sonnet-4-20250514}
    ports:
      - "8003:8003"
    depends_on:
      - platform

  gate:
    build:
      context: ..
      dockerfile: docker/Dockerfile.gate
    environment:
      JWT_SECRET: ${JWT_SECRET:-dev_secret}
      PLATFORM_URL: http://platform:8001
      FLOW_URL: http://flow:8002
      RUNTIME_URL: http://runtime:8003
    ports:
      - "8000:8000"
    depends_on:
      - platform
      - flow
      - runtime

  chat:
    build:
      context: ..
      dockerfile: docker/Dockerfile.chat
    ports:
      - "3001:80"
    depends_on:
      - gate

  # ── Phase 2 服务 ──
  console:
    build:
      context: ..
      dockerfile: docker/Dockerfile.console
    ports:
      - "3002:80"
    depends_on:
      - gate

  gitea:
    image: gitea/gitea:latest
    environment:
      GITEA__database__DB_TYPE: postgres
      GITEA__database__HOST: postgres:5432
      GITEA__database__NAME: gitea
      GITEA__database__USER: quantalithos
      GITEA__database__PASSWD: ${DB_PASSWORD:-dev_password}
    ports:
      - "3000:3000"
      - "2222:22"
    volumes:
      - gitea_data:/data
    depends_on:
      - postgres

volumes:
  pgdata:
  miniodata:
  gitea_data:
```

## 三、启动脚本

```bash
#!/bin/bash
# scripts/setup-dev.sh — 一键启动开发环境

set -e

echo "🚀 Quantalithos AI 开发环境启动"

# 1. 检查依赖
command -v docker >/dev/null 2>&1 || { echo "❌ 需要安装 Docker"; exit 1; }
command -v docker-compose >/dev/null 2>&1 || { echo "❌ 需要安装 Docker Compose"; exit 1; }

# 2. 复制环境变量
if [ ! -f .env ]; then
    cp config/.env.example .env
    echo "📝 已创建 .env，请填入 LLM_API_KEY"
fi

# 3. 启动基础设施
echo "📦 启动 PostgreSQL + Redis + MinIO..."
docker-compose -f docker/docker-compose.yml up -d postgres redis minio

# 4. 等待数据库就绪
echo "⏳ 等待数据库就绪..."
until docker-compose exec postgres pg_isready -U quantalithos; do
    sleep 1
done

# 5. 运行迁移
echo "🔧 运行数据库迁移..."
bash scripts/migrate.sh

# 6. 导入初始数据
echo "🌱 导入初始数据..."
bash scripts/seed.sh

# 7. 启动应用服务
echo "🚀 启动应用服务..."
docker-compose -f docker/docker-compose.yml up -d platform flow runtime gate chat

echo ""
echo "✅ 开发环境已就绪"
echo "   gate:     http://localhost:8000"
echo "   chat:     http://localhost:3000"
echo "   platform: http://localhost:8001"
echo "   flow:     http://localhost:8002"
echo "   runtime:  http://localhost:8003"
echo "   MinIO:    http://localhost:9001"
```

## 四、端口分配

|------|------|------|
| gate | 8000 | 外部唯一入口 |

| console (prod) | 3002 | Nginx（Phase 2） |

| console (dev) | 5174 | Vite dev server（Phase 2） |

| Redis | 6379 | 缓存/事件总线 |

# quantalithos-marketplace 详细设计

> 资产市场 — 发布/搜索/下载/导入，集成到 platform hub-marketplace 模块

---

## 一、模块架构

```
console 市场页面
      │
      │ sdk-api
      ▼
gate → platform hub-marketplace
            │
            ├── assets/          资产 CRUD + 版本管理
            ├── search/          搜索（标签 + 全文）
            ├── review/          评分 + 评论
            ├── download/        下载 + 统计
            └── import/          一键导入到项目
                  │
                  │ 根据 asset_type 分发
                  ▼
            hub-agent-config     （导入 process_template / role / rule）
            hub-design           （导入 component）
            MinIO                （下载 app 构建产物）
```

## 二、Rust 数据模型

```rust
// hub-marketplace/src/model.rs
//
// 资产市场数据模型 — 6 种资产类型的统一存储。
//
// 设计意图：
//   所有资产类型（流程模板/角色定义/共享规则/组件/技能包/应用）共用一张表，
//   通过 asset_type 字段区分。这样搜索和排行榜可以跨类型聚合。
//
// content_data vs file_path（互斥）：
//   结构化资产（process_template/role_definition/shared_rule/component）→ content_data（JSONB）
//   文件类资产（app）→ file_path（指向 MinIO/S3）
//   两者不会同时有值。
//
// status 生命周期：
//   draft → published（作者发布）
//   published → unlisted（作者下架，已安装的不受影响）
//   published → removed（管理员删除，违规内容）

#[derive(Debug, Clone, Serialize, Deserialize, sqlx::FromRow)]
pub struct MarketplaceAsset {
    pub id: i64,                         // 自增主键
    pub asset_type: String,              // 资产类型 — 合法值：
                                         //   "process_template" — 流程模板（BPMN 定义）
                                         //   "role_definition" — 角色定义（identity + tools + constraints）
                                         //   "shared_rule" — 共享规则（行为规范）
                                         //   "skill" — 技能包（prompt + 工具组合）
                                         //   "component" — UI 组件（template + styles + props）
                                         //   "app" — 应用作品（构建产物，Docker 镜像/静态文件）
    pub name: String,                    // 资产名称（如 "TDD 工作流"）
    pub description: String,             // 详细描述（Markdown）
    pub author_id: String,               // 作者 ID（用户 ID 或 "system"）
    pub author_name: String,             // 作者显示名称
    pub version: String,                 // 当前版本号（如 "1.2.0"）
    pub tags: serde_json::Value,         // 标签列表（JSON 数组，如 ["tdd", "testing", "workflow"]）
    pub category: String,                // 分类（如 "development" / "testing" / "deployment"）
    pub license: String,                 // 许可证（如 "MIT" / "Apache-2.0" / "proprietary"）
    pub content_type: String,            // 内容格式："yaml" | "json" | "zip" | "docker_image"
    pub content_data: Option<serde_json::Value>,  // 结构化内容（process_template/role/rule/component）
    pub file_path: Option<String>,       // 文件路径（app 类型，指向 MinIO）
    pub file_size: Option<i64>,          // 文件大小（字节）
    pub readme: String,                  // README 内容（Markdown，展示在详情页）
    pub screenshots: serde_json::Value,  // 截图 URL 列表（JSON 数组）
    pub download_count: i32,             // 下载/导入次数 — 用于排行榜
    pub rating_avg: f64,                 // 平均评分（1.0–5.0）— 用于排序
    pub rating_count: i32,               // 评分人数
    pub status: String,                  // 状态：draft | published | unlisted | removed
    pub created_at: DateTime<Utc>,
    pub updated_at: DateTime<Utc>,
}

/// 资产版本历史 — 每次更新创建新版本，旧版本保留
#[derive(Debug, Clone, Serialize, Deserialize, sqlx::FromRow)]
pub struct MarketplaceVersion {
    pub id: i64,
    pub asset_id: i64,                   // 所属资产 ID
    pub version: String,                 // 版本号（如 "1.1.0"）
    pub changelog: String,               // 更新日志（Markdown）
    pub content_data: Option<serde_json::Value>,  // 该版本的内容快照
    pub file_path: Option<String>,       // 该版本的文件路径
    pub created_at: DateTime<Utc>,
}

/// 用户评价 — Phase 3 实现
#[derive(Debug, Clone, Serialize, Deserialize, sqlx::FromRow)]
pub struct MarketplaceReview {
    pub id: i64,
    pub asset_id: i64,                   // 所属资产 ID
    pub user_id: String,                 // 评价者 ID
    pub rating: i32,                     // 评分（1–5 星）
    pub comment: String,                 // 评论内容
    pub created_at: DateTime<Utc>,
}
```

## 三、API Handler 实现

### 3.1 搜索

```rust
// hub-marketplace/src/handler.rs
//
// 资产搜索 — 支持按类型/关键词/标签/分类筛选，按下载量/评分/时间排序。
//
// 搜索策略：
//   Phase 2：PostgreSQL ILIKE 模糊匹配（小数据量足够）
//   Phase 3：Meilisearch 全文搜索（大数据量 + 中文分词）
//
// tags @> 操作符：
//   PostgreSQL JSONB 包含操作符 — 检查 tags 数组是否包含指定标签。
//   如 tags = ["tdd", "testing"]，查询 tags @> '["tdd"]' → true。
//
// 分页限制（page_size.min(100)）：
//   防止客户端请求过大的页面导致数据库压力。最大 100 条/页。

#[derive(Deserialize)]
pub struct SearchQuery {
    pub asset_type: Option<String>,      // 按资产类型筛选
    pub q: Option<String>,               // 关键词搜索（匹配 name + description）
    pub tag: Option<String>,             // 按标签筛选
    pub category: Option<String>,        // 按分类筛选
    pub sort: Option<String>,            // 排序方式：downloads | rating | newest（默认 newest）
    pub page: Option<i32>,               // 页码（从 1 开始）
    pub page_size: Option<i32>,          // 每页条数（默认 20，最大 100）
}

pub async fn search_assets(
    State(state): State<AppState>,
    Query(q): Query<SearchQuery>,
) -> Result<Json<PaginatedResult<MarketplaceAsset>>, PlatformError> {
    let mut query = sqlx::QueryBuilder::new("SELECT * FROM marketplace_assets WHERE status = 'published'");

    if let Some(t) = &q.asset_type {
        query.push(" AND asset_type = ").push_bind(t);
    }
    if let Some(keyword) = &q.q {
        query.push(" AND (name ILIKE ").push_bind(format!("%{}%", keyword))
             .push(" OR description ILIKE ").push_bind(format!("%{}%", keyword)).push(")");
    }
    if let Some(tag) = &q.tag {
        query.push(" AND tags @> ").push_bind(serde_json::json!([tag]));
    }

    match q.sort.as_deref() {
        Some("downloads") => query.push(" ORDER BY download_count DESC"),
        Some("rating") => query.push(" ORDER BY rating_avg DESC"),
        _ => query.push(" ORDER BY created_at DESC"),
    };

    // 分页
    let page = q.page.unwrap_or(1);
    let size = q.page_size.unwrap_or(20).min(100);
    query.push(" LIMIT ").push_bind(size).push(" OFFSET ").push_bind((page - 1) * size);

    let assets = query.build_query_as::<MarketplaceAsset>()
        .fetch_all(&state.db).await?;

    Ok(Json(PaginatedResult { items: assets, page, page_size: size }))
}
```

### 3.2 一键导入

```rust
// hub-marketplace/src/handler.rs
//
// 一键导入 — 将市场资产导入到用户的项目中。
//
// 设计意图：
//   用户在市场浏览到感兴趣的资产后，点击"导入到项目"即可使用。
//   根据 asset_type 分发到不同的目标表：
//     process_template → hub-agent-config.process_templates
//     role_definition → hub-agent-config.role_definitions
//     shared_rule → hub-agent-config.shared_rules
//     component → hub-design.components
//     app → 返回下载链接（Runner 下载运行，不入库）
//
// 导入后的 scope：
//   导入的资产 scope="project"，只对目标项目生效，不影响其他项目。
//   用户可以在导入后修改（fork 语义）。
//
// increment_download 的调用时机：
//   导入成功后才计数（失败不计），确保统计准确。

pub async fn import_to_project(
    State(state): State<AppState>,
    Path((asset_id, project_id)): Path<(i64, String)>,
) -> Result<Json<ImportResult>, PlatformError> {
    let asset = state.marketplace_repo.get(asset_id).await?
        .ok_or(PlatformError::NotFound("Asset not found".into()))?;

    let result = match asset.asset_type.as_str() {
        "process_template" => {
            let data = asset.content_data.ok_or(PlatformError::ValidationError("No content".into()))?;
            state.agent_config_repo.insert_process_template(&project_id, "project", data).await?;
            ImportResult { imported_to: "process_templates".into(), success: true }
        },
        "role_definition" => {
            let data = asset.content_data.ok_or(PlatformError::ValidationError("No content".into()))?;
            state.agent_config_repo.insert_role(&project_id, "project", data).await?;
            ImportResult { imported_to: "role_definitions".into(), success: true }
        },
        "shared_rule" => {
            let data = asset.content_data.ok_or(PlatformError::ValidationError("No content".into()))?;
            state.agent_config_repo.insert_rule(&project_id, "project", data).await?;
            ImportResult { imported_to: "shared_rules".into(), success: true }
        },
        "component" => {
            let data = asset.content_data.ok_or(PlatformError::ValidationError("No content".into()))?;
            state.design_repo.insert_component(&project_id, "project", data).await?;
            ImportResult { imported_to: "components".into(), success: true }
        },
        "app" => {
            // app 类型返回下载链接，不导入数据库
            ImportResult { imported_to: "download".into(), success: true }
        },
        _ => return Err(PlatformError::ValidationError(format!("Unknown asset_type: {}", asset.asset_type))),
    };

    // 记录下载统计
    state.marketplace_repo.increment_download(asset_id).await?;

    Ok(Json(result))
}
```

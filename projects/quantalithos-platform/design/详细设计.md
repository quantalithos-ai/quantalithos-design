### 3.1 workitem 状态机校验

```rust
// hub-workitem/src/state_machine.rs

/// 状态机校验 — 检查状态流转是否合法
pub fn validate_transition(item_type: &str, from: &str, to: &str) -> Result<(), String> {
    let allowed = match item_type {
        "task" => TASK_TRANSITIONS,
        "requirement" => REQUIREMENT_TRANSITIONS,
        "bug" => BUG_TRANSITIONS,
        _ => return Err(format!("Unknown item_type: {}", item_type)),
    };

    if allowed.contains(&(from, to)) {
        Ok(())
    } else {
        Err(format!("Invalid transition: {} → {} for {}", from, to, item_type))
    }
}

const TASK_TRANSITIONS: &[(&str, &str)] = &[
    ("created", "assigned"),
    ("assigned", "in_progress"),
    ("in_progress", "in_review"),
    ("in_review", "in_testing"),
    ("in_review", "review_rejected"),
    ("in_testing", "completed"),
    ("in_testing", "bug_found"),
    ("review_rejected", "in_progress"),
    ("bug_found", "in_progress"),
];

const BUG_TRANSITIONS: &[(&str, &str)] = &[
    ("open", "confirmed"),
    ("open", "rejected"),
    ("open", "cannot_reproduce"),
    ("confirmed", "fixing"),
    ("fixing", "fixed"),
    ("fixing", "blocked"),
    ("blocked", "fixing"),
    ("fixed", "verifying"),
    ("verifying", "closed"),
    ("verifying", "reopen"),
    ("reopen", "confirmed"),
];

const REQUIREMENT_TRANSITIONS: &[(&str, &str)] = &[
    ("draft", "reviewing"),
    ("reviewing", "approved"),
    ("reviewing", "needs_revision"),
    ("needs_revision", "reviewing"),
    ("approved", "in_progress"),
    ("in_progress", "implemented"),
    ("implemented", "verified"),
    ("implemented", "verify_failed"),
    ("verify_failed", "in_progress"),
    ("verified", "closed"),
];
```

## 四、事件总线

```rust
// hub-common/src/event_bus.rs

use tokio::sync::broadcast;

#[derive(Clone, Debug)]
pub struct PlatformEvent {
    pub event_type: String,           // "WorkItemCreated" | "WorkItemAssigned" | ...
    pub project_id: String,
    pub data: serde_json::Value,
    pub timestamp: chrono::DateTime<chrono::Utc>,
}

pub struct EventBus {
    sender: broadcast::Sender<PlatformEvent>,
}

impl EventBus {
    pub fn new() -> Self {
        let (sender, _) = broadcast::channel(1024);
        Self { sender }
    }

    pub fn publish(&self, event: PlatformEvent) {
        let _ = self.sender.send(event);
    }

    pub fn subscribe(&self) -> broadcast::Receiver<PlatformEvent> {
        self.sender.subscribe()
    }
}
```

## 五、三级覆盖实现

```rust
// hub-agent-config/src/scope_resolver.rs

/// 三级覆盖：project > company > system
pub async fn resolve_role(
    db: &PgPool,
    role_id: &str,
    project_id: Option<&str>,
) -> Result<RoleDefinition, Error> {
    // 1. 项目级（最高优先级）
    if let Some(pid) = project_id {
        if let Some(role) = find_role(db, role_id, "project", Some(pid)).await? {
            return Ok(role);
        }
    }

    // 2. 公司级
    if let Some(role) = find_role(db, role_id, "company", None).await? {
        return Ok(role);
    }

    // 3. 系统级（最低优先级）
    find_role(db, role_id, "system", None).await?
        .ok_or(Error::NotFound(format!("Role {} not found", role_id)))
}
```

```rust
// hub-workitem/src/handler.rs

#[derive(Deserialize)]
pub struct TransitionRequest {
    pub to_status: String,
    pub triggered_by: String,
    pub reason: Option<String>,
}

pub async fn transition(
    State(state): State<AppState>,
    Path(id): Path<String>,
    Json(req): Json<TransitionRequest>,
) -> Result<Json<WorkItem>, PlatformError> {
    let mut item = state.workitem_repo.get(&id).await?
        .ok_or(PlatformError::NotFound(format!("WorkItem {}", id)))?;

    let from_status = item.status.clone();

    // 1. 状态机校验
    validate_transition(&item.item_type, &from_status, &req.to_status)?;

    // 2. 自动指派
    auto_assign(&mut item, &req.to_status, &state).await?;

    // 3. 更新状态
    item.status = req.to_status.clone();
    item.updated_at = Utc::now();
    state.workitem_repo.update(&item).await?;

    // 4. 记录 transition
    state.workitem_repo.insert_transition(WorkItemTransition {
        id: 0,
        workitem_id: id.clone(),
        from_status: from_status.clone(),
        to_status: req.to_status.clone(),
        triggered_by: req.triggered_by.clone(),
        reason: req.reason.clone(),
        assignee_before: None,
        assignee_after: item.assignee_id.clone(),
        created_at: Utc::now(),
    }).await?;

    // 5. 发布事件
    state.event_bus.publish(PlatformEvent {
        event_type: "WorkItemStatusChanged".into(),
        project_id: item.project_id.clone(),
        data: serde_json::json!({
            "workitem_id": id,
            "from_status": from_status,
            "to_status": req.to_status,
            "triggered_by": req.triggered_by,
        }),
        timestamp: Utc::now(),
    });

    // 6. 如果新状态是 assigned，额外发布 WorkItemAssigned
    if req.to_status == "assigned" {
        state.event_bus.publish(PlatformEvent {
            event_type: "WorkItemAssigned".into(),
            project_id: item.project_id.clone(),
            data: serde_json::json!({
                "workitem_id": id,
                "assignee_id": item.assignee_id,
                "item_type": item.item_type,
            }),
            timestamp: Utc::now(),
        });
    }

    // 7. 如果任务完成，检查是否解锁下游
    if req.to_status == "completed" {
        state.task_scheduler.on_task_completed(&id).await?;
    }

    Ok(Json(item))
}
```

```rust
// hub-workitem/src/task_scheduler.rs

pub struct TaskScheduler {
    workitem_repo: Arc<WorkItemRepo>,
    event_bus: Arc<EventBus>,
}

impl TaskScheduler {
    /// TL 触发执行：检查依赖图，无依赖的立即 assigned
    pub async fn start_execution(&self, project_id: &str) -> Result<StartExecutionResult, PlatformError> {
        let items = self.workitem_repo.list_by_project(project_id).await?;
        let deps = self.workitem_repo.list_dependencies(project_id).await?;

        // 构建依赖图：workitem_id → [depends_on_id, ...]
        let dep_map = self.build_dep_map(&deps);

        let mut assigned_count = 0;
        for item in &items {
            if item.status != "created" { continue; }

            // 检查所有依赖是否已完成
            let blocked = dep_map.get(&item.id)
                .map(|deps| deps.iter().any(|d| {
                    items.iter().find(|i| i.id == *d)
                        .map(|i| i.status != "completed")
                        .unwrap_or(true)
                }))
                .unwrap_or(false);

            if !blocked {
                // 无依赖或依赖已满足 → assigned
                self.workitem_repo.update_status(&item.id, "assigned").await?;
                self.auto_assign_on_created(&item).await?;

                self.event_bus.publish(PlatformEvent {
                    event_type: "WorkItemAssigned".into(),
                    project_id: project_id.into(),
                    data: serde_json::json!({
                        "workitem_id": item.id,
                        "assignee_id": item.assignee_id,
                        "item_type": item.item_type,
                    }),
                    timestamp: Utc::now(),
                });
                assigned_count += 1;
            }
        }

        Ok(StartExecutionResult { assigned_count })
    }

    /// 任务完成时：检查是否解锁下游任务
    pub async fn on_task_completed(&self, workitem_id: &str) -> Result<(), PlatformError> {
        let item = self.workitem_repo.get(workitem_id).await?
            .ok_or(PlatformError::NotFound(workitem_id.into()))?;

        // 找到所有依赖此任务的下游
        let downstream = self.workitem_repo
            .find_dependents(workitem_id).await?;

        for d in downstream {
            if d.status != "created" { continue; }

            // 检查该下游的所有依赖是否都已完成
            let all_deps_met = self.workitem_repo
                .check_all_deps_completed(&d.id).await?;

            if all_deps_met {
                self.workitem_repo.update_status(&d.id, "assigned").await?;
                self.auto_assign_on_created(&d).await?;

                self.event_bus.publish(PlatformEvent {
                    event_type: "TaskBecameReady".into(),
                    project_id: item.project_id.clone(),
                    data: serde_json::json!({
                        "workitem_id": d.id,
                        "assignee_id": d.assignee_id,
                        "unblocked_by": workitem_id,
                    }),
                    timestamp: Utc::now(),
                });
            }
        }

        // 检查项目内所有任务是否全部完成
        let all_done = self.workitem_repo
            .all_tasks_completed(&item.project_id).await?;
        if all_done {
            self.event_bus.publish(PlatformEvent {
                event_type: "AllTasksCompleted".into(),
                project_id: item.project_id.clone(),
                data: serde_json::json!({}),
                timestamp: Utc::now(),
            });
        }

        Ok(())
    }

    /// 波次调度：按依赖层级分组
    pub async fn wave_schedule(&self, project_id: &str) -> Result<Vec<Vec<String>>, PlatformError> {
        let items = self.workitem_repo.list_by_project(project_id).await?;
        let deps = self.workitem_repo.list_dependencies(project_id).await?;
        let dep_map = self.build_dep_map(&deps);

        let mut waves: Vec<Vec<String>> = vec![];
        let mut scheduled: HashSet<String> = HashSet::new();

        loop {
            let wave: Vec<String> = items.iter()
                .filter(|i| !scheduled.contains(&i.id))
                .filter(|i| {
                    dep_map.get(&i.id)
                        .map(|ds| ds.iter().all(|d| scheduled.contains(d)))
                        .unwrap_or(true)
                })
                .map(|i| i.id.clone())
                .collect();

            if wave.is_empty() { break; }
            for id in &wave { scheduled.insert(id.clone()); }
            waves.push(wave);
        }

        Ok(waves)
    }
}
```

### 10.2 自动指派完整规则

```rust
// hub-workitem/src/service.rs

pub async fn auto_assign(
    item: &mut WorkItem,
    new_status: &str,
    state: &AppState,
) -> Result<(), PlatformError> {
    let members = state.project_repo.list_members(&item.project_id).await?;

    let find_by_role = |role: &str| -> Option<String> {
        members.iter()
            .find(|m| m.role_id == role && m.status == "active")
            .map(|m| m.agent_id.clone())
    };

    match (item.item_type.as_str(), new_status) {
        // Task: in_review → 指派给 TL
        ("task", "in_review") => {
            if let Some(tl) = find_by_role("tech_lead") {
                item.assignee_id = Some(tl);
            }
        },
        // Task: in_testing → 指派给 tester
        ("task", "in_testing") => {
            if let Some(qa) = find_by_role("tester") {
                item.assignee_id = Some(qa);
            }
        },
        // Task: review_rejected / bug_found → 回到原开发者
        ("task", "review_rejected") | ("task", "bug_found") => {
            // assignee 保持不变（还是原开发者）
        },
        // Bug: fixed → 指派给 reporter 验证
        ("bug", "fixed") => {
            item.verifier_id = Some(item.reporter_id.clone());
            item.assignee_id = Some(item.reporter_id.clone());
        },
        // Bug: reopen → 指派给上一个修复者（assignee 不变）
        ("bug", "reopen") => {},
        _ => {}
    }

    Ok(())
}
```

## 十二、错误处理

```rust
// hub-common/src/error.rs

use axum::http::StatusCode;
use axum::response::{IntoResponse, Response};

#[derive(Debug)]
pub enum PlatformError {
    NotFound(String),
    InvalidTransition(String),
    DuplicateKey(String),
    ValidationError(String),
    Unauthorized(String),
    InternalError(String),
}

impl IntoResponse for PlatformError {
    fn into_response(self) -> Response {
        let (status, code, message) = match &self {
            Self::NotFound(msg)          => (StatusCode::NOT_FOUND, "NOT_FOUND", msg.clone()),
            Self::InvalidTransition(msg) => (StatusCode::BAD_REQUEST, "INVALID_TRANSITION", msg.clone()),
            Self::DuplicateKey(msg)      => (StatusCode::CONFLICT, "DUPLICATE_KEY", msg.clone()),
            Self::ValidationError(msg)   => (StatusCode::BAD_REQUEST, "VALIDATION_ERROR", msg.clone()),
            Self::Unauthorized(msg)      => (StatusCode::UNAUTHORIZED, "UNAUTHORIZED", msg.clone()),
            Self::InternalError(msg)     => (StatusCode::INTERNAL_SERVER_ERROR, "INTERNAL_ERROR", msg.clone()),
        };

        let body = serde_json::json!({
            "error": {
                "code": code,
                "message": message,
            }
        });

        (status, axum::Json(body)).into_response()
    }
}

impl From<sqlx::Error> for PlatformError {
    fn from(e: sqlx::Error) -> Self {
        match e {
            sqlx::Error::RowNotFound => Self::NotFound("Record not found".into()),
            sqlx::Error::Database(ref db_err) if db_err.code().as_deref() == Some("23505") => {
                Self::DuplicateKey(db_err.message().to_string())
            },
            _ => Self::InternalError(e.to_string()),
        }
    }
}
```

错误码映射：

| 错误 | HTTP 状态码 | 错误码 | 触发场景 |
|------|-----------|--------|---------|
| NotFound | 404 | NOT_FOUND | 工单/项目/角色不存在 |
| InvalidTransition | 400 | INVALID_TRANSITION | 状态机校验失败（如 created → completed） |
| DuplicateKey | 409 | DUPLICATE_KEY | 重复创建（如同一 agent 重复加入项目） |
| ValidationError | 400 | VALIDATION_ERROR | 请求参数校验失败 |
| Unauthorized | 401 | UNAUTHORIZED | 认证头缺失或无效 |
| InternalError | 500 | INTERNAL_ERROR | 数据库异常、未预期错误 |

## 十三、完整示例：博客项目 TASK-003 标签系统

```
① TL 创建任务
   POST /api/projects/blog/workitems
   { item_type: "task", title: "文章标签系统", reporter_id: "tech-lead",
     priority: "high", description: "实现标签 CRUD + 文章关联" }
   → 201 Created, workitem_id = "TASK-003", status = "created"
   → 发布 WorkItemCreated 事件

② TL 设置依赖
   POST /api/workitems/TASK-003/dependencies
   { depends_on_id: "TASK-001" }  // 依赖数据库表结构任务
   → TASK-001 已 completed，依赖满足

③ TL 触发执行
   POST /api/projects/blog/workitems/start-execution
   → TaskScheduler 检查依赖图
   → TASK-003 无未完成依赖 → status 改为 "assigned"
   → auto_assign：item_type=task, 保持 assignee 为 null（由 flow 指定）
   → 发布 WorkItemAssigned 事件
     { workitem_id: "TASK-003", assignee_id: null, item_type: "task" }

④ flow 监听到 WorkItemAssigned
   → 构造 FlowMessage(type="node_execute", agent_id="backend-dev-1",
       payload={ node: {node_type: "abstract_task", ...}, control_level: "enforced", process_id: "tdd_workflow" })
   → 发送给 runtime

⑤ runtime 执行
   → backend-dev-1 按 TDD 工作流执行
   → 完成后调用 workitem_transition 工具：
     POST /api/workitems/TASK-003/transition
     { to_status: "in_review", triggered_by: "backend-dev-1" }
   → 状态机校验：in_progress → in_review ✓
   → auto_assign：in_review → 指派给 tech-lead
   → 记录 transition
   → 发布 WorkItemStatusChanged + WorkItemAssigned

⑥ flow 监听到 WorkItemAssigned（TL 审查）
   → 构造 FlowMessage(type="node_execute", agent_id="tech-lead", node=AbstractTask("代码审查"))
   → TL 审查通过：
     POST /api/workitems/TASK-003/transition
     { to_status: "in_testing", triggered_by: "tech-lead" }
   → auto_assign：in_testing → 指派给 tester-1
   → 发布事件

⑦ tester-1 测试通过
   POST /api/workitems/TASK-003/transition
   { to_status: "completed", triggered_by: "tester-1" }
   → on_task_completed：检查下游依赖
   → 如果 TASK-004 依赖 TASK-003 且 TASK-003 是最后一个依赖
     → TASK-004 变为 assigned → 发布 TaskBecameReady
   → 如果项目内所有任务完成 → 发布 AllTasksCompleted
```

```sql
-- ============================================================
-- 1. projects

CREATE TABLE projects (
    id              VARCHAR(64)     PRIMARY KEY,
    name            VARCHAR(256)    NOT NULL,
    description     TEXT            NOT NULL DEFAULT '',
    status          VARCHAR(32)     NOT NULL DEFAULT 'active',
    pipeline_process_id VARCHAR(128) NOT NULL,
    created_at      TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
    updated_at      TIMESTAMPTZ     NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_projects_status ON projects(status);

CREATE TABLE project_members (
    id              BIGSERIAL       PRIMARY KEY,
    project_id      VARCHAR(64)     NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
    agent_id        VARCHAR(64)     NOT NULL,
    role_id         VARCHAR(64)     NOT NULL,
    status          VARCHAR(32)     NOT NULL DEFAULT 'active',
    joined_at       TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
    UNIQUE(project_id, agent_id)
);

CREATE INDEX idx_pm_project ON project_members(project_id);


-- ============================================================
-- 2. workitems
-- ============================================================

CREATE TABLE workitems (
    id              VARCHAR(64)     PRIMARY KEY,
    project_id      VARCHAR(64)     NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
    item_type       VARCHAR(32)     NOT NULL,       -- task | requirement | bug
    title           VARCHAR(512)    NOT NULL,
    description     TEXT            NOT NULL DEFAULT '',
    status          VARCHAR(32)     NOT NULL DEFAULT 'created',
    priority        VARCHAR(16)     NOT NULL DEFAULT 'medium',
    severity        VARCHAR(16),                    -- bug 专用
    assignee_id     VARCHAR(64),
    reporter_id     VARCHAR(64)     NOT NULL,
    verifier_id     VARCHAR(64),
    parent_id       VARCHAR(64)     REFERENCES workitems(id) ON DELETE SET NULL,
    labels          JSONB           NOT NULL DEFAULT '[]',
    created_at      TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
    updated_at      TIMESTAMPTZ     NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_wi_project ON workitems(project_id);
CREATE INDEX idx_wi_status ON workitems(project_id, status);
CREATE INDEX idx_wi_type ON workitems(project_id, item_type);
CREATE INDEX idx_wi_assignee ON workitems(assignee_id);

CREATE TABLE workitem_dependencies (
    id              BIGSERIAL       PRIMARY KEY,
    workitem_id     VARCHAR(64)     NOT NULL REFERENCES workitems(id) ON DELETE CASCADE,
    depends_on_id   VARCHAR(64)     NOT NULL REFERENCES workitems(id) ON DELETE CASCADE,
    UNIQUE(workitem_id, depends_on_id),
    CHECK(workitem_id != depends_on_id)
);

CREATE TABLE workitem_comments (
    id              BIGSERIAL       PRIMARY KEY,
    workitem_id     VARCHAR(64)     NOT NULL REFERENCES workitems(id) ON DELETE CASCADE,
    author_id       VARCHAR(64)     NOT NULL,
    content         TEXT            NOT NULL,
    created_at      TIMESTAMPTZ     NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_wc_item ON workitem_comments(workitem_id);

CREATE TABLE workitem_transitions (
    id              BIGSERIAL       PRIMARY KEY,
    workitem_id     VARCHAR(64)     NOT NULL REFERENCES workitems(id) ON DELETE CASCADE,
    from_status     VARCHAR(32)     NOT NULL,
    to_status       VARCHAR(32)     NOT NULL,
    triggered_by    VARCHAR(64)     NOT NULL,
    reason          TEXT,
    assignee_before VARCHAR(64),
    assignee_after  VARCHAR(64),
    created_at      TIMESTAMPTZ     NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_wt_item ON workitem_transitions(workitem_id);


-- ============================================================

    file_size       BIGINT,
    mime_type       VARCHAR(128),
    source_node_id  VARCHAR(128),
    task_ref        VARCHAR(64)     REFERENCES workitems(id) ON DELETE SET NULL,
    tags            JSONB           NOT NULL DEFAULT '[]',
    created_by      VARCHAR(64)     NOT NULL,
    created_at      TIMESTAMPTZ     NOT NULL DEFAULT NOW()

CREATE INDEX idx_art_project ON artifacts(project_id);
CREATE INDEX idx_art_type ON artifacts(project_id, artifact_type);
CREATE INDEX idx_art_task ON artifacts(task_ref);

CREATE TABLE artifact_versions (
    id              BIGSERIAL       PRIMARY KEY,
    artifact_id     VARCHAR(64)     NOT NULL REFERENCES artifacts(id) ON DELETE CASCADE,
    version         INT             NOT NULL,
    structured_data JSONB,
    file_path       VARCHAR(1024),
    file_size       BIGINT,
    created_by      VARCHAR(64)     NOT NULL,
```

## 三、状态机实现

### 3.1 workitem 状态机校验

```rust
// hub-workitem/src/state_machine.rs
//
// 工单状态机 — 定义所有合法的状态转换路径。
//
// 设计意图：
//   状态机是工单生命周期的"交通规则" — 只允许合法的状态跳转。
//   非法跳转（如 created → completed）会被 validate_transition 拒绝。
//
// 三种工单类型各有独立的状态图：
//   task：created → assigned → in_progress → in_review → in_testing → completed
//   bug：open → confirmed → fixing → fixed → verifying → closed
//   requirement：draft → reviewing → approved → in_progress → implemented → verified → closed
//
// 为什么用 const 数组而非 HashMap：
//   状态转换是编译时已知的固定集合（<20 个元素），数组 contains() 比 HashMap 更快。

/// 状态机校验 — 检查状态流转是否合法
pub fn validate_transition(item_type: &str, from: &str, to: &str) -> Result<(), String> {
    let allowed = match item_type {
        "task" => TASK_TRANSITIONS,
        "requirement" => REQUIREMENT_TRANSITIONS,
        "bug" => BUG_TRANSITIONS,
        _ => return Err(format!("Unknown item_type: {}", item_type)),
    };

    if allowed.contains(&(from, to)) {
        Ok(())
    } else {
        Err(format!("Invalid transition: {} → {} for {}", from, to, item_type))
    }
}

// Task 状态转换图：
//   created → assigned → in_progress → in_review → in_testing → completed
//   打回路径：in_review → review_rejected → in_progress
//   测试失败：in_testing → bug_found → in_progress
const TASK_TRANSITIONS: &[(&str, &str)] = &[
    ("created", "assigned"),
    ("assigned", "in_progress"),
    ("in_progress", "in_review"),
    ("in_review", "in_testing"),
    ("in_review", "review_rejected"),
    ("in_testing", "completed"),
    ("in_testing", "bug_found"),
    ("review_rejected", "in_progress"),
    ("bug_found", "in_progress"),
];

// Bug 状态转换图：
//   open → confirmed → fixing → fixed → verifying → closed
//   打回：verifying → reopen → confirmed
const BUG_TRANSITIONS: &[(&str, &str)] = &[
    ("open", "confirmed"),
    ("open", "rejected"),
    ("open", "cannot_reproduce"),
    ("confirmed", "fixing"),
    ("fixing", "fixed"),
    ("fixing", "blocked"),
    ("blocked", "fixing"),
    ("fixed", "verifying"),
    ("verifying", "closed"),
    ("verifying", "reopen"),
    ("reopen", "confirmed"),
];

// Requirement 状态转换图：
//   draft → reviewing → approved → in_progress → implemented → verified → closed
//   打回：reviewing → needs_revision → reviewing
//   验收失败：implemented → verify_failed → in_progress
const REQUIREMENT_TRANSITIONS: &[(&str, &str)] = &[
    ("draft", "reviewing"),
    ("reviewing", "approved"),
    ("reviewing", "needs_revision"),
    ("needs_revision", "reviewing"),
    ("approved", "in_progress"),
    ("in_progress", "implemented"),
    ("implemented", "verified"),
    ("implemented", "verify_failed"),
    ("verify_failed", "in_progress"),
    ("verified", "closed"),
];
```

### 3.2 自动指派规则

```rust
// hub-workitem/src/service.rs
//
// 自动指派（简化版，§3.2）— transition handler 内部调用。
// 完整版见 §10.2，此处是概要展示核心逻辑。
//
// find_by_role 闭包：从项目成员中找第一个匹配角色且 active 的 Agent。

/// 状态流转时自动变更 assignee（异步，需要查询项目成员）
pub async fn auto_assign(
    item: &mut WorkItem,
    new_status: &str,
    state: &AppState,
) -> Result<(), PlatformError> {
    let members = state.project_repo.list_members(&item.project_id).await?;

    let find_by_role = |role: &str| -> Option<String> {
        members.iter()
            .find(|m| m.role_id == role && m.status == "active")
            .map(|m| m.agent_id.clone())
    };

    match (item.item_type.as_str(), new_status) {
        // Task: in_review → 指派给 TL
        ("task", "in_review") => {
            if let Some(tl) = find_by_role("tech_lead") {
                item.assignee_id = Some(tl);
            }
        },
        // Task: in_testing → 指派给 tester
        ("task", "in_testing") => {
            if let Some(qa) = find_by_role("tester") {
                item.assignee_id = Some(qa);
            }
        },
        // Task: review_rejected / bug_found → 保持原开发者
        ("task", "review_rejected") | ("task", "bug_found") => {},
        // Bug: fixed → 指派给 reporter 验证
        ("bug", "fixed") => {
            item.verifier_id = Some(item.reporter_id.clone());
            item.assignee_id = Some(item.reporter_id.clone());
        },
        // Bug: reopen → assignee 不变
        ("bug", "reopen") => {},
        _ => {}
    }

    Ok(())
}
```

## 四、事件总线

```rust
// hub-common/src/event_bus.rs
//
// 事件总线 — platform 内部的异步事件分发机制。
//
// 设计意图：
//   platform 的各种状态变更（工单状态、任务调度、产物创建）需要通知外部消费者（flow/gate）。
//   EventBus 提供进程内的 pub/sub 机制，解耦事件生产者和消费者。
//
// Phase 1：tokio broadcast channel（进程内，单实例部署）
// Phase 2：Redis Pub/Sub（跨进程，支持多实例水平扩展）
//
// 容量设计（1024）：
//   broadcast channel 的 buffer 大小。超过 1024 条未消费的事件时，
//   最旧的事件会被丢弃（lagged）。1024 对于单项目足够（峰值约 100 事件/分钟）。
//   Phase 2 切换 Redis 后此限制消失。

use tokio::sync::broadcast;

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum PlatformEventType {
    // ── 项目事件 ──
    ProjectCreated,           // 新项目创建（Assistant 创建时触发）
    ProjectUpdated,           // 项目信息更新（名称/描述等）
    ProjectStatusChanged,     // 项目状态变更（active → suspended / completed / archived）
    ProjectStageChanged,      // 阶段推进（如 requirement → design）— chat 前端刷新进度条
    TeamMemberAdded,          // 新成员加入项目 — chat 前端显示通知
    TeamMemberRemoved,        // 成员移除

    // ── 工单事件 — 工单生命周期中的关键状态变更 ──
    WorkItemCreated,          // 工单创建（TL 或 Tester 创建）
    WorkItemAssigned,         // 工单被指派（自动指派或手动指派）— flow 监听此事件发 FlowMessage
    WorkItemStatusChanged,    // 工单状态变更（transition handler 触发）
    WorkItemVerified,         // 工单验证完成（Tester 通过/不通过）
    WorkItemClosed,           // 工单关闭（最终状态）
    WorkItemCommented,        // 工单被评论（Phase 2）

    // ── 任务调度事件 — 依赖图驱动的调度事件 ──
    ExecutionStarted,         // TL 触发任务调度（start_execution 调用时）
    TaskBecameReady,          // 任务依赖满足，变为可执行 — flow 监听此事件分配 Agent
    AllTasksCompleted,        // 项目内所有任务完成 — flow 监听此事件推进到验收阶段

    // ── 产物事件 ──
    ArtifactCreated,          // 产物创建（Agent 提交代码/文档后）
    ArtifactUpdated,          // 产物更新（新版本）

    // ── Agent 配置事件 ──
    RoleUpdated,              // 角色定义变更 — runtime 可能需要重新加载
    SharedRuleUpdated,        // 共享规则变更
    ProcessTemplateUpdated,   // 流程模板变更

    // ── 快照事件 ──
    AgentSnapshotCreated,     // Agent 快照保存（suspend 时）
    WorkflowSnapshotCreated,  // 工作流快照保存（flow 每步持久化时）
}

#[derive(Clone, Debug, Serialize)]
pub struct PlatformEvent {
    pub event_type: PlatformEventType,    // 事件类型 — 消费者按此字段过滤感兴趣的事件
    pub project_id: String,               // 所属项目 — 多项目隔离，消费者只关心自己项目的事件
    pub data: PlatformEventData,          // 类型化的事件数据 — 每种事件有不同的字段
    pub timestamp: chrono::DateTime<chrono::Utc>,  // 事件产生时间 — 用于排序和超时检测
}

/// 类型化的事件数据 — 每种事件有明确的字段
///
/// 为什么用 enum 而非统一的 HashMap/JSON：
///   1. 编译时类型安全 — 生产者不会遗漏必填字段
///   2. 消费者 match 时有穷举检查 — 新增事件类型时编译器提醒
///   3. 序列化后结构清晰 — 便于调试和日志分析
///
/// 为什么用 #[serde(untagged)]：
///   序列化时不包含 enum variant 名称，直接展开字段。
///   这样 JSON 格式更扁平，消费者（flow/gate）解析更简单。
#[derive(Clone, Debug, Serialize)]
#[serde(untagged)]
pub enum PlatformEventData {
    /// 工单被指派 — flow 的 PlatformListener.on_workitem_assigned() 消费
    WorkItemAssigned {
        workitem_id: String,              // 被指派的工单 ID
        assignee_id: Option<String>,      // 新负责人 Agent ID（None = 待 flow 指定）
        item_type: String,                // 工单类型（task/bug）— flow 据此决定 control_level
    },
    /// 工单状态变更 — gate 转发给前端实时更新 UI
    WorkItemStatusChanged {
        workitem_id: String,
        from_status: String,              // 变更前状态
        to_status: String,                // 变更后状态
        triggered_by: String,             // 触发者（Agent ID 或 "system"）
    },
    /// 任务依赖满足 — flow 监听后为该任务分配 Agent
    TaskBecameReady {
        workitem_id: String,              // 变为 ready 的工单 ID
        assignee_id: Option<String>,      // 预分配的负责人（可能为 None）
        unblocked_by: String,             // 是哪个任务完成后解锁了此任务
    },
    /// 所有任务完成 — flow 监听后推进 Pipeline 到验收阶段
    AllTasksCompleted,
    /// 通用简单事件 — 只需要一个 ID 的事件（Created/Updated 等）
    /// 消费者通过 event_type 区分具体是什么事件，再按 id 查询详情
    Simple {
        id: String,                       // 资源 ID（工单 ID / 产物 ID / 快照 ID）
    },
}

/// 事件总线 — 进程内 pub/sub（Phase 1）
///
/// 设计职责：
///   维护一个 broadcast channel，生产者调用 publish() 发送事件，
///   消费者调用 subscribe() 获取 Receiver 接收事件。
///
/// 线程安全：
///   broadcast::Sender 是 Clone + Send + Sync，可以安全地在多个 handler 间共享。
///   通常包装在 Arc<EventBus> 中放入 AppState。
pub struct EventBus {
    sender: broadcast::Sender<PlatformEvent>,  // broadcast channel 的发送端
}

impl EventBus {
    pub fn new() -> Self {
        // 1024 = channel buffer 大小，超过后旧事件被丢弃（lagged error）
        let (sender, _) = broadcast::channel(1024);
        Self { sender }
    }

    /// 发布事件 — 所有 subscriber 都会收到
    /// let _ = 忽略发送失败（没有 subscriber 时 send 会返回 Err，这是正常的）
    pub fn publish(&self, event: PlatformEvent) {
        let _ = self.sender.send(event);
    }

    /// 订阅事件 — 返回一个 Receiver，调用者 await recv() 接收事件
    pub fn subscribe(&self) -> broadcast::Receiver<PlatformEvent> {
        self.sender.subscribe()
    }
}
```

## 五、三级覆盖实现

```rust
// hub-agent-config/src/scope_resolver.rs
//
// 三级覆盖解析器 — 实现 system → company → project 的配置覆盖机制。
//
// 设计意图：
//   角色定义、共享规则、流程模板都支持三级覆盖：
//   system（系统预装）→ company（公司统一）→ project（项目特化）
//   高优先级覆盖低优先级，runtime/flow 收到的是最终合并结果。
//
// 查找策略（高优先级优先）：
//   1. 先查 project 级（scope="project" AND project_id=指定值）
//   2. 没有 → 查 company 级（scope="company"，不需要 project_id）
//   3. 没有 → 查 system 级（scope="system"，兜底）
//   4. 都没有 → 返回 NotFound 错误
//
// 为什么不做合并（merge）而是直接覆盖：
//   Phase 1 简化实现 — 高优先级的定义完整替换低优先级。
//   Phase 2 引入 extends 机制后，支持字段级合并（如 project 级只覆盖 tools 字段）。

/// 三级覆盖：project > company > system
pub async fn resolve_role(
    db: &PgPool,
    role_id: &str,
    project_id: Option<&str>,
) -> Result<RoleDefinition, Error> {
    // 1. 项目级（最高优先级）— 项目特化的角色定义
    if let Some(pid) = project_id {
        if let Some(role) = find_role(db, role_id, "project", Some(pid)).await? {
            return Ok(role);
        }
    }

    // 2. 公司级 — 公司统一的角色定义（不需要 project_id，因为公司级对所有项目生效）
    if let Some(role) = find_role(db, role_id, "company", None).await? {
        return Ok(role);
    }

    // 3. 系统级（最低优先级）— 系统预装的默认角色定义
    find_role(db, role_id, "system", None).await?
        .ok_or(Error::NotFound(format!("Role {} not found", role_id)))
}
```

```rust
// hub-workitem/src/model.rs
//
// 工单数据模型 — 项目中的最小可执行工作单元。
//
// 设计说明：
//   status 字段用 String 而非 enum 的原因：
//   三种 item_type 有不同的状态集（task 有 9 种状态，bug 有 11 种，requirement 有 10 种），
//   如果用 enum 需要一个巨大的 union enum，且新增状态时要改 Rust 代码。
//   用 String + validate_transition() 校验更灵活。

//   parent_id 的用途：

    pub project_id: String,           // 所属项目
    pub item_type: String,            // 工单类型 — 合法值：
                                      //   "task" — 开发任务（TL 拆解后分配给开发者）
                                      //   "requirement" — 需求（用户提出，TL 评审后拆解为 task）
                                      //   "bug" — 缺陷（Tester 发现，指派给开发者修复）

    pub description: String,          // 详细描述（Agent 执行时作为 action 注入 prompt）
    pub status: String,               // 状态机驱动 — 合法值按 item_type 区分：
                                      //   task: created | assigned | in_progress | in_review |
                                      //         review_rejected | in_testing | bug_found | completed
                                      //   bug: open | confirmed | rejected | cannot_reproduce |
                                      //        fixing | blocked | fixed | verifying | reopen | closed
                                      //   requirement: draft | reviewing | needs_revision | approved |
                                      //               in_progress | implemented | verified | verify_failed | closed
    pub priority: String,             // 优先级 — 合法值：
                                      //   "critical" — 紧急，阻塞其他工作

    pub parent_id: Option<String>,    // 父工单（用于 epic → task 拆分）
    pub labels: serde_json::Value,    // 标签列表（JSONB），用于筛选和分类

    pub updated_at: DateTime<Utc>,

/// 工单状态变更记录 — 审计追溯用（谁在什么时候做了什么变更）
///
/// 每次 transition handler 执行成功后插入一条记录。
/// 前端的"工单时间线"视图从此表查询。
```

```rust
// hub-integration/src/git_handler.rs
//
// Git 集成 handler — 接收 Gitea webhook + 查询工单关联的 Git 信息。
//
// 两类端点：
//   1. Webhook 接收（POST /api/webhooks/gitea）— push/PR 事件回调
//   2. 关联查询（GET /api/workitems/:id/git-refs）— 工单关联的 commits 和 PRs

/// 接收 Gitea webhook
pub async fn handle_gitea_webhook(
    State(state): State<AppState>,
    Json(payload): Json<serde_json::Value>,
) -> Result<StatusCode, PlatformError> {
    let event_type = payload["action"].as_str().unwrap_or("");

    match event_type {
        // push 事件：解析 commits，关联工单
        "push" => {
            let commits = payload["commits"].as_array().unwrap_or(&vec![]);
            for commit in commits {
                let message = commit["message"].as_str().unwrap_or("");
                let workitem_id = parse_workitem_ref(message);
                state.git_repo.insert_commit(GitCommitRef {
                    id: 0,
                    repo_id: resolve_repo_id(&state, &payload).await?,
                    commit_sha: commit["id"].as_str().unwrap_or("").into(),
                    message: message.into(),
                    author_agent_id: resolve_author(&commit),
                    workitem_id,
                    created_at: Utc::now(),
                }).await?;
            }
            Ok(StatusCode::OK)
        },
        // PR 事件：记录 PR 信息
        "opened" | "closed" | "merged" => {
            let pr = &payload["pull_request"];
            state.git_repo.upsert_pr(GitPullRequest {
                id: 0,
                repo_id: resolve_repo_id(&state, &payload).await?,
                gitea_pr_number: pr["number"].as_i64().unwrap_or(0),
                title: pr["title"].as_str().unwrap_or("").into(),
                source_branch: pr["head"]["ref"].as_str().unwrap_or("").into(),
                target_branch: pr["base"]["ref"].as_str().unwrap_or("").into(),
                status: event_type.into(),
                author_agent_id: resolve_author(&pr["user"]),
                workitem_id: parse_workitem_ref(pr["title"].as_str().unwrap_or("")),
                created_at: Utc::now(),
                merged_at: if event_type == "merged" { Some(Utc::now()) } else { None },
            }).await?;
            Ok(StatusCode::OK)
        },
        _ => Ok(StatusCode::OK),
    }
}

/// 查询工单关联的 commits 和 PRs — console 工单详情页使用
pub async fn workitem_git_refs(
    State(state): State<AppState>,
    Path(workitem_id): Path<String>,
) -> Result<Json<WorkItemGitRefs>, PlatformError> {
    let commits = state.git_repo.find_commits_by_workitem(&workitem_id).await?;
    let prs = state.git_repo.find_prs_by_workitem(&workitem_id).await?;

}

/// 从 commit message 解析工单引用
/// 支持格式：feat(TASK-003): ... / fix(BUG-002): ... / [TASK-003] ...
fn parse_workitem_ref(message: &str) -> Option<String> {
    let re = regex::Regex::new(r"(TASK|BUG|REQ)-\d+").unwrap();
    re.find(message).map(|m| m.as_str().to_string())
}
```

```rust
// hub-design/src/handler.rs

// 设计系统 handler — Design Tokens + 组件库的 CRUD。
// 原型编辑器的 handler 在 Phase 3 补充。

/// 组件库列表（系统级 + 项目级合并，三级覆盖）
pub async fn list_components(
    State(state): State<AppState>,
    Query(params): Query<ComponentQuery>,
) -> Result<Json<Vec<Component>>, PlatformError> {
    let system_components = state.design_repo.list_components("system", None).await?;
    let project_components = if let Some(pid) = &params.project_id {
        state.design_repo.list_components("project", Some(pid)).await?
    } else {
        vec![]
    };

    // 合并：项目级覆盖系统级（按 component_id 去重）
    let mut merged = system_components;
    for pc in project_components {
        if let Some(pos) = merged.iter().position(|c| c.component_id == pc.component_id) {
            merged[pos] = pc;  // 项目级覆盖
        } else {
            merged.push(pc);   // 项目级新增
        }
    }

    if let Some(cat) = &params.category {
        merged.retain(|c| &c.category == cat);
    }
    Ok(Json(merged))
}
```

```rust
// hub-workitem/src/handler.rs
//
// 工单状态转换 — platform 最核心的 API handler。
//
// 设计意图：
//   工单状态转换是整个系统的"心跳" — 每次 Agent 完成一步工作，
//   都通过 transition 推进工单状态，触发后续的自动指派和事件通知。
//
// 7 步流程设计理由：
//   1. 状态机校验 — 防止非法跳转（如 created → completed）
//   2. 自动指派 — 减少人工干预（in_review 自动指派给 TL）
//   3. 更新状态 — 持久化到数据库
//   4. 记录 transition — 审计追溯（谁在什么时候做了什么）
//   5. 发布状态变更事件 — 通知 gate 实时更新前端
//   6. 发布指派事件 — 通知 flow 为新负责人发 FlowMessage
//   7. 检查下游解锁 — 依赖图驱动的任务调度
//
// 幂等性：
//   如果 from_status == to_status，validate_transition 会拒绝（不允许自转换）。
//   重复调用不会产生副作用。

#[derive(Deserialize)]
pub struct TransitionRequest {
    pub to_status: String,           // 目标状态（如 "in_review"）
    pub triggered_by: String,        // 触发者（Agent ID 或 "system"）
    pub reason: Option<String>,      // 变更原因（可选，用于审计日志）
}

pub async fn transition(
    State(state): State<AppState>,
    Path(id): Path<String>,
    Json(req): Json<TransitionRequest>,
) -> Result<Json<WorkItem>, PlatformError> {
    let mut item = state.workitem_repo.get(&id).await?
        .ok_or(PlatformError::NotFound(format!("WorkItem {}", id)))?;

    let from_status = item.status.clone();
    let assignee_before = item.assignee_id.clone();

    // 1. 状态机校验
    validate_transition(&item.item_type, &from_status, &req.to_status)?;

    // 2. 自动指派
    auto_assign(&mut item, &req.to_status, &state).await?;

    // 3. 更新状态
    item.status = req.to_status.clone();
    item.updated_at = Utc::now();
    state.workitem_repo.update(&item).await?;

    // 4. 记录 transition
    state.workitem_repo.insert_transition(WorkItemTransition {
        id: 0,
        workitem_id: id.clone(),
        from_status: from_status.clone(),
        to_status: req.to_status.clone(),
        triggered_by: req.triggered_by.clone(),
        reason: req.reason.clone(),
        assignee_before: assignee_before,
        assignee_after: item.assignee_id.clone(),
        created_at: Utc::now(),
    }).await?;

    // 5. 发布事件（类型化）
    state.event_bus.publish(PlatformEvent {
        event_type: PlatformEventType::WorkItemStatusChanged,
        project_id: item.project_id.clone(),
        data: PlatformEventData::WorkItemStatusChanged {
            workitem_id: id.clone(),
            from_status: from_status.clone(),
            to_status: req.to_status.clone(),
            triggered_by: req.triggered_by.clone(),
        },
        timestamp: Utc::now(),
    });

    // 6. 如果新状态是 assigned，额外发布 WorkItemAssigned
    if req.to_status == "assigned" {
        state.event_bus.publish(PlatformEvent {
            event_type: PlatformEventType::WorkItemAssigned,
            project_id: item.project_id.clone(),
            data: PlatformEventData::WorkItemAssigned {
                workitem_id: id.clone(),
                assignee_id: item.assignee_id.clone(),
                item_type: item.item_type.clone(),
            },
            timestamp: Utc::now(),
        });
    }

    // 7. 如果任务完成，检查是否解锁下游
    if req.to_status == "completed" {
        state.task_scheduler.on_task_completed(&id).await?;
    }

    Ok(Json(item))
}
```

### 10.1 任务调度 Service

```rust
// hub-workitem/src/task_scheduler.rs
//
// 任务调度器 — 依赖图驱动的工单调度核心逻辑。
//
// 设计意图：
//   TL 拆解任务后定义依赖关系（如"标签系统"依赖"文章模型"），
//   TaskScheduler 负责：
//   1. start_execution：初始启动时，找出所有无依赖的任务立即 assigned
//   2. on_task_completed：某任务完成后，检查是否解锁了下游任务
//   3. wave_schedule：计算波次（拓扑排序），用于前端依赖图可视化
//
// 调度算法：
//   本质是 DAG（有向无环图）的拓扑排序 + 逐层释放。
//   无依赖的任务 = 入度为 0 的节点 = 第一波。
//   第一波完成后，检查哪些节点的入度变为 0 = 第二波。
//
// 幂等性保证：
//   只处理 status="created" 的工单，已经 assigned/in_progress 的跳过。
//   重复调用 start_execution 不会重复指派。

pub struct TaskScheduler {
    workitem_repo: Arc<WorkItemRepo>,    // 工单数据访问层
    event_bus: Arc<EventBus>,            // 事件总线 — 发布调度事件通知 flow
}

impl TaskScheduler {
    /// TL 触发执行：检查依赖图，无依赖的立即 assigned
    ///
    /// 调用时机：TL 完成任务拆解后，调用 POST /workitems/start-execution
    /// 返回：被 assigned 的任务数量
    pub async fn start_execution(&self, project_id: &str) -> Result<StartExecutionResult, PlatformError> {
        let items = self.workitem_repo.list_by_project(project_id).await?;
        let deps = self.workitem_repo.list_dependencies(project_id).await?;

        // 构建依赖图：workitem_id → [depends_on_id, ...]
        let dep_map = self.build_dep_map(&deps);

        let mut assigned_count = 0;
        for item in &items {
            // 幂等：只处理 created 状态的工单，已 assigned 的跳过
            if item.status != "created" { continue; }

            // 检查所有依赖是否已完成
            let blocked = dep_map.get(&item.id)
                .map(|deps| deps.iter().any(|d| {
                    items.iter().find(|i| i.id == *d)
                        .map(|i| i.status != "completed")
                        .unwrap_or(true)
                }))
                .unwrap_or(false);

            if !blocked {
                // 无依赖或依赖已满足 → assigned
                self.workitem_repo.update_status(&item.id, "assigned").await?;
                self.auto_assign_on_created(&item).await?;

                self.event_bus.publish(PlatformEvent {
                    event_type: PlatformEventType::WorkItemAssigned,
                    project_id: project_id.into(),
                    data: PlatformEventData::WorkItemAssigned {
                        workitem_id: item.id.clone(),
                        assignee_id: item.assignee_id.clone(),
                        item_type: item.item_type.clone(),
                    },
                    timestamp: Utc::now(),
                });
                assigned_count += 1;
            }
        }

        Ok(StartExecutionResult { assigned_count })
    }

    /// 任务完成时：检查是否解锁下游任务
    pub async fn on_task_completed(&self, workitem_id: &str) -> Result<(), PlatformError> {
        let item = self.workitem_repo.get(workitem_id).await?
            .ok_or(PlatformError::NotFound(workitem_id.into()))?;

        // 找到所有依赖此任务的下游
        let downstream = self.workitem_repo
            .find_dependents(workitem_id).await?;

        for d in downstream {
            if d.status != "created" { continue; }

            // 检查该下游的所有依赖是否都已完成
            let all_deps_met = self.workitem_repo
                .check_all_deps_completed(&d.id).await?;

            if all_deps_met {
                self.workitem_repo.update_status(&d.id, "assigned").await?;
                self.auto_assign_on_created(&d).await?;

                self.event_bus.publish(PlatformEvent {
                    event_type: PlatformEventType::TaskBecameReady,
                    project_id: item.project_id.clone(),
                    data: PlatformEventData::TaskBecameReady {
                        workitem_id: d.id.clone(),
                        assignee_id: d.assignee_id.clone(),
                        unblocked_by: workitem_id.to_string(),
                    },
                    timestamp: Utc::now(),
                });
            }
        }

        // 检查项目内所有任务是否全部完成
        let all_done = self.workitem_repo
            .all_tasks_completed(&item.project_id).await?;
        if all_done {
            self.event_bus.publish(PlatformEvent {
                event_type: PlatformEventType::AllTasksCompleted,
                project_id: item.project_id.clone(),
                data: PlatformEventData::AllTasksCompleted,
                timestamp: Utc::now(),
            });
        }

        Ok(())
    }

    /// 波次调度：按依赖层级分组
    pub async fn wave_schedule(&self, project_id: &str) -> Result<Vec<Vec<String>>, PlatformError> {
        let items = self.workitem_repo.list_by_project(project_id).await?;
        let deps = self.workitem_repo.list_dependencies(project_id).await?;
        let dep_map = self.build_dep_map(&deps);

        let mut waves: Vec<Vec<String>> = vec![];
        let mut scheduled: HashSet<String> = HashSet::new();

        loop {
            let wave: Vec<String> = items.iter()
                .filter(|i| !scheduled.contains(&i.id))
                .filter(|i| {
                    dep_map.get(&i.id)
                        .map(|ds| ds.iter().all(|d| scheduled.contains(d)))
                        .unwrap_or(true)
                })
                .map(|i| i.id.clone())
                .collect();

            if wave.is_empty() { break; }
            for id in &wave { scheduled.insert(id.clone()); }
            waves.push(wave);
        }

        Ok(waves)
    }
}
```

### 10.2 自动指派完整规则

```rust
// hub-workitem/src/service.rs
//
// 自动指派 — 工单状态变更时自动切换负责人。
//
// 设计意图：
//   减少人工干预 — 开发完成后自动指派给 TL 审查，审查通过后自动指派给 Tester。
//   Agent 不需要手动调用 workitem_assign，transition 时自动处理。
//
// 指派规则（按 item_type + new_status 组合）：
//   task + in_review → TL（代码审查）
//   task + in_testing → Tester（功能测试）
//   task + review_rejected / bug_found → 保持原开发者（打回重做）
//   bug + fixed → reporter（谁报的 bug 谁验证）
//   bug + reopen → 保持原修复者（重新修复）
//
// find_by_role 的查找策略：
//   从项目成员中找第一个匹配角色且 status=active 的 Agent。
//   Phase 2 扩展：支持负载均衡（多个同角色 Agent 时选最空闲的）。

pub async fn auto_assign(
    item: &mut WorkItem,
    new_status: &str,
    state: &AppState,
) -> Result<(), PlatformError> {
    let members = state.project_repo.list_members(&item.project_id).await?;

    let find_by_role = |role: &str| -> Option<String> {
        members.iter()
            .find(|m| m.role_id == role && m.status == "active")
            .map(|m| m.agent_id.clone())
    };

    match (item.item_type.as_str(), new_status) {
        // Task: in_review → 指派给 TL
        ("task", "in_review") => {
            if let Some(tl) = find_by_role("tech_lead") {
                item.assignee_id = Some(tl);
            }
        },
        // Task: in_testing → 指派给 tester
        ("task", "in_testing") => {
            if let Some(qa) = find_by_role("tester") {
                item.assignee_id = Some(qa);
            }
        },
        // Task: review_rejected / bug_found → 回到原开发者
        ("task", "review_rejected") | ("task", "bug_found") => {
            // assignee 保持不变（还是原开发者）
        },
        // Bug: fixed → 指派给 reporter 验证
        ("bug", "fixed") => {
            item.verifier_id = Some(item.reporter_id.clone());
            item.assignee_id = Some(item.reporter_id.clone());
        },
        // Bug: reopen → 指派给上一个修复者（assignee 不变）
        ("bug", "reopen") => {},
        _ => {}
    }

    Ok(())
}
```

### 11.1 ViewRenderer trait

```rust
// hub-api/src/views.rs
//
// 多视图渲染 — 同一份数据按不同消费者需求输出不同格式。
//
// 设计意图：
//   同一个 WorkItem，Agent 需要结构化的 facts/checks/scores（做决策用），
//   前端需要 title/status/badges（展示用），两者需求完全不同。
//   ViewRenderer trait 让每种资源类型实现多种视图格式，
//   API 层按 URL 路径选择调用哪个 render 方法。
//
// 为什么用 trait 而非在 handler 中直接构造 JSON：
//   1. 视图逻辑可能很复杂（需要关联查询、计算分数），独立出来便于测试
//   2. 多个资源类型共享相同的视图接口（统一的 render_ai / render_card）
//   3. Phase 2 可以引入缓存层（对 render 结果缓存）
//
// 三种 trait 的分层：
//   ViewRenderer — 所有资源都有 AI 视图和 Card 视图
//   ProjectViewRenderer — 只有 Project 有 Progress 视图（Pipeline 进度）
//   ProcessViewRenderer — 只有 ProcessTemplate 有 Graph 视图（BPMN 图）

/// 通用视图渲染 trait — 所有资源类型实现
pub trait ViewRenderer {
    type Entity;

    /// AI 视图 — Agent 消费，结构化的 facts + checks + scores + actions
    fn render_ai(&self, entity: &Self::Entity) -> serde_json::Value;
    /// Card 视图 — 前端列表展示，title + status + badges + actions
    fn render_card(&self, entity: &Self::Entity) -> serde_json::Value;
}

/// 项目特有视图 — 只有 Project 实现
pub trait ProjectViewRenderer: ViewRenderer {
    /// Progress 视图 — Pipeline 六阶段进度 + 当前阶段流程图
    fn render_progress(&self, entity: &Self::Entity) -> serde_json::Value;
}

/// 流程模板特有视图 — 只有 ProcessTemplate 实现
pub trait ProcessViewRenderer: ViewRenderer {
    /// Graph 视图 — BPMN 节点 + 边 + 布局坐标（前端流程编辑器用）
    fn render_graph(&self, entity: &Self::Entity) -> serde_json::Value;
}
```

## 十二、错误处理

```rust
// hub-common/src/error.rs
//
// 统一错误类型 — platform 所有 handler 的错误返回格式。
//
// 设计意图：
//   1. 所有 API 返回统一的错误 JSON 格式：{ error: { code, message } }
//   2. 通过 IntoResponse trait 自动将错误转为 HTTP 响应（axum 框架约定）
//   3. 通过 From<sqlx::Error> 自动将数据库错误转为业务错误（? 操作符直接用）
//
// 错误分类设计：
//   - 客户端错误（4xx）：NotFound / InvalidTransition / DuplicateKey / ValidationError / Unauthorized
//   - 服务端错误（5xx）：InternalError
//   客户端错误 = 调用者可以修正（换个 ID / 换个状态 / 补充参数）
//   服务端错误 = 调用者无法修正（数据库挂了 / 未预期的 bug）
//
// From<sqlx::Error> 的映射逻辑：
//   RowNotFound → NotFound（查询结果为空）
//   Database code "23505" → DuplicateKey（PostgreSQL 唯一约束冲突）
//   其他 → InternalError（未预期的数据库错误）

use axum::http::StatusCode;
use axum::response::{IntoResponse, Response};

#[derive(Debug)]
pub enum PlatformError {
    NotFound(String),            // 资源不存在 — 工单/项目/角色/快照找不到
    InvalidTransition(String),   // 非法状态转换 — 状态机校验失败（如 created → completed）
    DuplicateKey(String),        // 重复创建 — 唯一约束冲突（如同一 agent 重复加入项目）
    ValidationError(String),     // 参数校验失败 — 请求体缺少必填字段或格式错误
    Unauthorized(String),        // 认证失败 — token 缺失/过期/无效
    InternalError(String),       // 服务端错误 — 数据库异常、未预期的 panic
}

impl IntoResponse for PlatformError {
    fn into_response(self) -> Response {
        let (status, code, message) = match &self {
            Self::NotFound(msg)          => (StatusCode::NOT_FOUND, "NOT_FOUND", msg.clone()),
            Self::InvalidTransition(msg) => (StatusCode::BAD_REQUEST, "INVALID_TRANSITION", msg.clone()),
            Self::DuplicateKey(msg)      => (StatusCode::CONFLICT, "DUPLICATE_KEY", msg.clone()),
            Self::ValidationError(msg)   => (StatusCode::BAD_REQUEST, "VALIDATION_ERROR", msg.clone()),
            Self::Unauthorized(msg)      => (StatusCode::UNAUTHORIZED, "UNAUTHORIZED", msg.clone()),
            Self::InternalError(msg)     => (StatusCode::INTERNAL_SERVER_ERROR, "INTERNAL_ERROR", msg.clone()),
        };

        // 统一的错误响应格式 — 所有消费者（sdk/flow/runtime）按此格式解析
        let body = serde_json::json!({
            "error": {
                "code": code,
                "message": message,
            }
        });

        (status, axum::Json(body)).into_response()
    }
}

/// 从 sqlx 数据库错误自动转换 — 让 handler 中可以直接用 ? 操作符
impl From<sqlx::Error> for PlatformError {
    fn from(e: sqlx::Error) -> Self {
        match e {
            // 查询结果为空 → 404
            sqlx::Error::RowNotFound => Self::NotFound("Record not found".into()),
            // PostgreSQL 唯一约束冲突（error code 23505）→ 409
            sqlx::Error::Database(ref db_err) if db_err.code().as_deref() == Some("23505") => {
                Self::DuplicateKey(db_err.message().to_string())
            },
            // 其他数据库错误 → 500（不暴露内部细节给客户端，但记录日志）
            _ => Self::InternalError(e.to_string()),
        }
    }
}
```

错误码映射：

| 错误 | HTTP 状态码 | 错误码 | 触发场景 |
|------|-----------|--------|---------|
| NotFound | 404 | NOT_FOUND | 工单/项目/角色不存在 |
| InvalidTransition | 400 | INVALID_TRANSITION | 状态机校验失败（如 created → completed） |
| DuplicateKey | 409 | DUPLICATE_KEY | 重复创建（如同一 agent 重复加入项目） |
| ValidationError | 400 | VALIDATION_ERROR | 请求参数校验失败 |
| Unauthorized | 401 | UNAUTHORIZED | 认证头缺失或无效 |
| InternalError | 500 | INTERNAL_ERROR | 数据库异常、未预期错误 |

## 十三点六、可观测性设计

### 13.6.1 技术选型

| 维度 | 选型 | 说明 |
|------|------|------|
| 日志 | tracing + tracing-subscriber | Rust 生态标准，结构化 JSON 日志 |
| 追踪 | tracing + opentelemetry-otlp | 分布式追踪，request_id 贯穿全链路 |
| 指标 | metrics + prometheus-exporter | /metrics 端点，Prometheus 格式 |
| 可视化 | Grafana（Phase 2 部署） | Phase 1 用日志 + CLI 查询 |

### 13.6.2 日志规范

```rust
// 每个 handler 用 #[tracing::instrument] 自动记录入参和耗时
#[tracing::instrument(skip(state), fields(request_id = %req_id))]
pub async fn transition(
    State(state): State<AppState>,
    Path(id): Path<String>,
    Json(req): Json<TransitionRequest>,
) -> Result<Json<WorkItem>, PlatformError> {
    tracing::info!(workitem_id = %id, to_status = %req.to_status, "transition requested");
    // ... 业务逻辑
    tracing::info!(workitem_id = %id, "transition completed");
    Ok(Json(item))
}
```

### 13.6.3 关键指标

| 指标名 | 类型 | 说明 |
|--------|------|------|
| `platform_http_requests_total` | Counter | 请求总数（method/path/status） |
| `platform_http_request_duration_seconds` | Histogram | 请求耗时分布 |
| `platform_workitem_transitions_total` | Counter | 状态转换次数（from/to） |
| `platform_events_published_total` | Counter | 事件发布总数（event_type） |
| `platform_db_query_duration_seconds` | Histogram | 数据库查询耗时 |

### 13.6.4 request_id 追踪链

```
gate（生成 X-Request-Id）→ platform（tracing span）→ 事件（携带 request_id）
  → flow（继承）→ FlowMessage（携带）→ runtime（继承）

一个 request_id 贯穿整条链路，可按 request_id 过滤完整调用链。
```

## 十五、性能边界与容量规划

### 15.1 Phase 1 目标（单机部署）

| 指标 | 目标值 | 说明 |
|------|--------|------|
| 并发项目数 | 1–5 | Phase 1 单用户使用，同时活跃 1–5 个项目 |
| 单项目工单数 | < 100 | 中小型项目（博客系统约 5–20 个工单） |
| 单项目 Agent 数 | < 10 | TL + 2 BE + 1 FE + 1 Tester + 1 DevOps |
| API 响应时间 P95 | < 100ms | 简单 CRUD（无复杂聚合查询） |
| API 响应时间 P95 | < 500ms | 复杂查询（views/progress 需要聚合多表） |
| 事件延迟 | < 50ms | 进程内 broadcast channel，几乎无延迟 |
| 数据库连接池 | 10 | PgPool max_connections=10（单机足够） |
| 内存占用 | < 256MB | Rust 二进制 + 连接池 + 事件 buffer |

### 15.2 Phase 2 目标（多用户 / 多项目）

| 指标 | 目标值 | 说明 |
|------|--------|------|
| 并发项目数 | 50–100 | 多用户同时使用 |
| 单项目工单数 | < 1000 | 大型项目 |
| API QPS | 500 | 读写混合（读 80% / 写 20%） |
| 事件延迟 | < 200ms | Redis Pub/Sub 跨进程 |
| 数据库连接池 | 50 | 多实例共享 PostgreSQL |
| 水平扩展 | 2–4 实例 | 无状态 API server + 共享 PostgreSQL + Redis |

### 15.3 已知瓶颈与应对

| 瓶颈 | Phase 1 应对 | Phase 2 应对 |
|------|-------------|-------------|
| 数据库查询 | 索引覆盖（已在 SQL DDL 中定义） | 读写分离 + 查询缓存 |
| 事件广播 | broadcast channel（1024 buffer） | Redis Pub/Sub（无 buffer 限制） |
| 文件存储 | 本地文件系统 | MinIO（S3 兼容，支持大文件流式传输） |
| views/progress 聚合 | 单次查询多表 JOIN | 预计算 + 缓存（Redis TTL 30s） |
| 全文搜索 | LIKE 模糊匹配（小数据量可接受） | PostgreSQL tsvector 或 Meilisearch |

### 15.4 数据量估算（单项目）

```
博客项目（典型中小型）：
  workitems: 5–20 条
  workitem_transitions: 50–200 条（每个工单约 5–10 次状态变更）
  workitem_dependencies: 5–15 条
  artifacts: 10–30 个（PRD + 设计文档 + 代码 + 测试报告）
  artifact_versions: 20–60 条（每个产物平均 2–3 个版本）
  team_members: 5–8 个
  agent_snapshots: 5–8 个（每个 Agent 一个）
  workflow_snapshots: 10–50 个（每步持久化）

大型项目（电商平台）：
  workitems: 100–500 条
  workitem_transitions: 1000–5000 条
  artifacts: 100–300 个
  team_members: 10–20 个

结论：Phase 1 单机 PostgreSQL 完全够用，无需分库分表。
```

### 3.1 AppManager 实现

```rust
// engine/app_manager.rs

pub struct AppManager {
    apps: HashMap<String, InstalledApp>,
    storage_root: PathBuf,                // "/apps/"
    platform_client: PlatformClient,
}

impl AppManager {
    /// 从 platform 拉取可安装的应用列表
    pub async fn fetch_available_apps(&self) -> Result<Vec<AvailableApp>, RunnerError> {
        let projects = self.platform_client.projects().list_with_builds().await?;
        Ok(projects.into_iter().map(|p| AvailableApp {
            project_id: p.id,
            name: p.name,
            description: p.description,
            latest_version: p.latest_build.map(|b| b.version),
            installed_version: self.apps.get(&p.id).map(|a| a.current_version.clone()),
            has_update: self.check_update(&p.id, &p.latest_build),
        }).collect())
    }

    /// 安装应用
    pub async fn install(&mut self, project_id: &str) -> Result<InstalledApp, RunnerError> {
        // 1. 获取最新构建信息
        let build = self.platform_client.builds()
            .get_latest(project_id).await?
            .ok_or(RunnerError::NoBuild(project_id.into()))?;

        // 2. 下载应用包
        let bundle_path = self.download(&build).await?;

        // 3. 校验 checksum
        let actual_hash = sha256_file(&bundle_path)?;
        if actual_hash != build.checksum {
            return Err(RunnerError::ChecksumMismatch {
                expected: build.checksum,
                actual: actual_hash,
            });
        }

        // 4. 解压到应用目录
        let install_path = self.storage_root.join(project_id);
        extract_zip(&bundle_path, &install_path)?;

        // 5. 读取 manifest
        let manifest = read_manifest(&install_path)?;

        // 6. 注册应用
        let app = InstalledApp {
            project_id: project_id.into(),
            name: manifest.name,
            description: manifest.description,
            current_version: build.version,
            build_id: build.build_id,
            build_type: manifest.build_type,
            install_path: install_path.to_string_lossy().into(),
            install_size: dir_size(&install_path)?,
            status: AppStatus::Ready,
            permissions: vec![],
            installed_at: Utc::now(),
            ..Default::default()
        };
        self.apps.insert(project_id.into(), app.clone());
        self.save_registry()?;

        Ok(app)
    }

    /// 检查更新
    pub async fn check_updates(&mut self) -> Result<Vec<UpdateInfo>, RunnerError> {
        let mut updates = vec![];
        for (pid, app) in &self.apps {
            let latest = self.platform_client.builds()
                .get_latest(pid).await?;
            if let Some(build) = latest {
                if build.version != app.current_version {
                    updates.push(UpdateInfo {
                        project_id: pid.clone(),
                        current_version: app.current_version.clone(),
                        new_version: build.version,
                        size: build.total_size,
                    });
                }
            }
        }
        Ok(updates)
    }
}
```

```rust
// engine/webview_container.rs

pub struct WebViewContainer {
    app_id: String,
    config: WebViewConfig,
    bridge: NativeBridge,
}

impl WebViewContainer {
    /// 创建并启动 WebView
    pub fn launch(app: &InstalledApp) -> Result<Self, RunnerError> {
        let config = WebViewConfig {
            app_id: app.project_id.clone(),
            entry_url: format!("file://{}/index.html", app.install_path),
            sandbox: SandboxConfig {
                storage_path: format!("{}/data/", app.install_path),
                storage_quota_mb: 100,
                network_whitelist: vec![
                    "localhost".into(),
                    "quantalithos.local".into(),
                ],
                allow_local_storage: true,
                allow_indexed_db: true,
                allow_cookies: true,
            },
            bridge_permissions: app.permissions.clone(),
            allow_navigation: vec![],
        };

        let bridge = NativeBridge::new(&app.project_id, &config.bridge_permissions);

        // 创建 WebView（平台相关）
        #[cfg(target_os = "macos")]
        let webview = create_wkwebview(&config, &bridge)?;
        #[cfg(target_os = "windows")]
        let webview = create_webview2(&config, &bridge)?;
        #[cfg(target_arch = "wasm32")]
        let webview = create_iframe(&config, &bridge)?;

        Ok(Self { app_id: app.project_id.clone(), config, bridge })
    }
}
```

### 4.2 原生桥接

```rust
// engine/bridge.rs

pub struct NativeBridge {
    app_id: String,
    permissions: Vec<Permission>,
}

impl NativeBridge {
    /// 注入到 WebView 的 JS 全局对象
    pub fn inject_js(&self) -> String {
        format!(r#"
            window.quantalithos = {{
                appId: "{}",
                
                // 存储（始终可用，沙箱隔离）
                storage: {{
                    get: (key) => window.__bridge_call("storage.get", {{ key }}),
                    set: (key, value) => window.__bridge_call("storage.set", {{ key, value }}),
                    remove: (key) => window.__bridge_call("storage.remove", {{ key }}),
                }},
                
                // 相机（需要授权）
                camera: {{
                    capture: () => window.__bridge_call("camera.capture", {{}}),
                    isAvailable: () => {},
                }},
                
                // 地理位置（需要授权）
                geolocation: {{
                    getCurrentPosition: () => window.__bridge_call("geo.current", {{}}),
                }},
                
                // 通知（需要授权）
                notification: {{
                    show: (title, body) => window.__bridge_call("notify.show", {{ title, body }}),
                }},
                
                // 剪贴板（需要授权）
                clipboard: {{
                    read: () => window.__bridge_call("clipboard.read", {{}}),
                    write: (text) => window.__bridge_call("clipboard.write", {{ text }}),
                }},
                
                // 应用信息（始终可用）
                app: {{
                    getVersion: () => "{}",
                    getPlatform: () => window.__bridge_call("app.platform", {{}}),
                }},
            }};
        "#, self.app_id, "0.3")
    }

    /// 处理 JS 调用
    pub async fn handle_call(&self, method: &str, args: serde_json::Value) -> Result<serde_json::Value, RunnerError> {
        // 权限检查
        let required = match method {
            "camera.capture" => Some(Permission::Camera),
            "geo.current" => Some(Permission::Geolocation),
            "notify.show" => Some(Permission::Notification),
            "clipboard.read" | "clipboard.write" => Some(Permission::Clipboard),
            "storage.get" | "storage.set" | "storage.remove" => None, // 存储始终可用
            "app.platform" => None,
            _ => return Err(RunnerError::UnknownBridgeMethod(method.into())),
        };

        if let Some(perm) = required {
            if !self.permissions.contains(&perm) {
                return Err(RunnerError::PermissionDenied(perm));
            }
        }

        // 执行原生操作
        match method {
            "storage.get" => {
                let key = args["key"].as_str().unwrap_or("");
                let value = self.storage_get(key)?;
                Ok(serde_json::json!(value))
            },
            "storage.set" => {
                let key = args["key"].as_str().unwrap_or("");
                let value = args["value"].as_str().unwrap_or("");
                self.storage_set(key, value)?;
                Ok(serde_json::json!(true))
            },
            "camera.capture" => {
                let photo = self.capture_photo().await?;
                Ok(serde_json::json!({ "data": photo.base64, "width": photo.width, "height": photo.height }))
            },
            "geo.current" => {
                let pos = self.get_position().await?;
                Ok(serde_json::json!({ "lat": pos.lat, "lng": pos.lng, "accuracy": pos.accuracy }))
            },
            _ => Err(RunnerError::UnknownBridgeMethod(method.into())),
        }
    }
}
```

### 5.2 VersionManager

```rust
// engine/version_manager.rs

pub struct VersionManager {
    app_manager: Arc<AppManager>,
    platform_client: PlatformClient,
}

impl VersionManager {
    /// 更新应用到最新版本
    pub async fn update(&self, project_id: &str) -> Result<InstalledApp, RunnerError> {
        let app = self.app_manager.get(project_id)?;
        let old_path = app.install_path.clone();
        let old_version = app.current_version.clone();

        // 1. 下载新版本到临时目录
        let new_build = self.platform_client.builds()
            .get_latest(project_id).await?
            .ok_or(RunnerError::NoBuild(project_id.into()))?;

        let temp_path = format!("{}.updating", old_path);
        self.app_manager.download_to(&new_build, &temp_path).await?;

        // 2. 校验
        let hash = sha256_file(&format!("{}.zip", temp_path))?;
        if hash != new_build.checksum {
            fs::remove_dir_all(&temp_path).ok();
            return Err(RunnerError::ChecksumMismatch {
                expected: new_build.checksum,
                actual: hash,
            });
        }

        // 3. 备份旧版本（用于回滚）
        let backup_path = format!(".backup-{}", old_path, old_version);
        fs::rename(&old_path, &backup_path)?;

        // 4. 替换
        fs::rename(&temp_path, &old_path)?;

        // 5. 更新注册信息
        let updated = self.app_manager.update_version(
            project_id, &new_build.version, &new_build.build_id
        )?;

        // 6. 清理备份（保留最近 2 个版本用于回滚）
        self.cleanup_old_backups(project_id, 2)?;

        Ok(updated)
    }

    /// 回滚到上一个版本
    pub async fn rollback(&self, project_id: &str) -> Result<InstalledApp, RunnerError> {
        let app = self.app_manager.get(project_id)?;
        let install_path = &app.install_path;

        // 查找最近的备份
        let backup = self.find_latest_backup(project_id)?
            .ok_or(RunnerError::NoBackup(project_id.into()))?;

        // 交换当前版本和备份
        let temp = format!("{}.rollback-temp", install_path);
        fs::rename(install_path, &temp)?;
        fs::rename(&backup.path, install_path)?;
        fs::rename(&temp, &backup.path)?;

        // 更新注册信息
        let rolled_back = self.app_manager.update_version(
            project_id, &backup.version, &backup.build_id
        )?;

        Ok(rolled_back)
    }
}
```

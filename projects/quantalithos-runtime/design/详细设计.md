> **设计原则：runtime 不自带任何模板**
>
> runtime 是纯执行引擎，不包含角色定义、共享规则、工作流模板。
> 所有配置从外部加载：
> - Phase 1：从本地文件加载（quantalithos-sync restore 拉取到本地）
>   - `project_files/{project}/agents/role_store/`
>   - `project_files/{project}/agents/shared_rules/`
>   - `project_files/{project}/agents/workflows/`
> - Phase 2：从 platform API 加载
>   - `GET /api/agent-config/roles/{id}`
>   - `GET /api/agent-config/shared-rules/{id}`
>   - `GET /api/agent-config/processes/{id}`
>
> 初始模板（系统角色、共享规则、标准流程）存储在 platform 数据库中，
> 由 infra 的初始数据脚本导入。runtime 自己不存储任何模板。

from pydantic import BaseModel
from typing import Optional, List
from datetime import datetime
from enum import Enum


class AssignmentMode(str, Enum):
    SOLO = "solo"
    COLLABORATION = "collaboration"


class NodeAssignment(BaseModel):
    """flow → runtime：节点执行分配"""
    assignment_id: str
    project_id: str
    process_instance_id: str
    node_id: str
    mode: AssignmentMode

    # solo 模式
    agent_id: Optional[str] = None
    action: Optional[str] = None
    call_process_id: Optional[str] = None

    # collaboration 模式
    moderator_id: Optional[str] = None
    participant_ids: List[str] = []
    agenda: Optional[str] = None

    # 共享
    input_refs: List[str] = []
    allowed_tools: List[str] = []
    output_schema: Optional[dict] = None
    created_at: datetime


class ResultType(str, Enum):
    COMPLETED = "completed"
    BLOCKED = "blocked"
    FAILED = "failed"


class NodeResult(BaseModel):
    """runtime → flow：节点执行结果"""
    assignment_id: str
    agent_id: str
    result_type: ResultType
    output: Optional[dict] = None
    reason: Optional[str] = None
    created_at: datetime
```

### 3.1 NodeAssignment 处理流程

```

    │
    ▼

    │
    ├── mode = "solo"
    │       │
    │       ▼

    │   4. tools/registry.py → 注册可用工具
    │      ├── 角色定义的 tools ∩ assignment.allowed_tools

    │      └── 注册 ask_teammate
    │   5. 检查 call_process_id

### 3.2 think→act 循环（AgentExecutor）

```python
# engine/executor.py

class AgentExecutor:
    async def run(self, agent: Agent) -> dict:
        """think→act 循环"""
        round_count = 0

        while round_count < self.max_rounds:
            round_count += 1

            # 1. think — 调用 LLM
            response = await self.llm_client.chat(
                LLMRequest(
                    model=self.config.llm_model,
                    messages=agent.conversation,
                    tools=self.tool_registry.get_schemas(agent.available_tools),
                )
            )

            # 2. 判断 LLM 的响应
            if response.tool_calls:
                # LLM 想调用工具
                for tool_call in response.tool_calls:
                    # 记录 assistant 消息（含 tool_calls）
                    agent.conversation.append(Message(
                        role="assistant",
                        content=response.content or "",
                        tool_calls=response.tool_calls,
                    ))

                    # act — 执行工具
                    result = await self.tool_executor.execute(
                        agent=agent,
                        tool_call=tool_call,
                    )

                    # 记录 tool 结果
                    agent.conversation.append(Message(
                        role="tool",
                        content=result.output,
                        tool_call_id=tool_call.id,
                    ))

                    # 检查是否调用了 submit_step_result
                    if tool_call.name == "submit_step_result":
                        return json.loads(tool_call.arguments)

            else:
                # LLM 只回复文本，没有调用工具
                agent.conversation.append(Message(
                    role="assistant",
                    content=response.content,
                ))
                # 继续循环（等待 LLM 决定调用工具或 submit）

        # 超过最大轮次
        raise MaxRoundsExceeded(f"Agent {agent.agent_id} exceeded {self.max_rounds} rounds")
```

### 3.3 工具权限校验

```python
# tools/executor.py

class ToolExecutor:
    async def execute(self, agent: Agent, tool_call: ToolCall) -> ToolResult:
        """执行工具调用（含权限校验）"""

        # 1. 检查工具是否在可用列表中
        if tool_call.name not in agent.available_tools:
            return ToolResult(
                tool_call_id=tool_call.id,
                success=False,
                output="",
                error=f"Tool '{tool_call.name}' is not available for this agent",
            )

        # 2. 获取工具定义
        tool = self.registry.get(tool_call.name)

        # 3. 路径权限校验（file_read/file_write 等）
        if tool.category == "builtin" and "path" in tool_call.arguments:
            path = tool_call.arguments["path"]
            if not self.check_path_permission(agent, tool_call.name, path):
                return ToolResult(
                    tool_call_id=tool_call.id,
                    success=False,
                    output="",
                    error=f"Permission denied: {path}",
                )

        # 4. 沙箱执行
        if tool.requires_sandbox:
            return await self.sandbox_execute(tool, tool_call)

        # 5. 正常执行
        try:
            output = await tool.handler(**tool_call.arguments)
            return ToolResult(
                tool_call_id=tool_call.id,
                success=True,
                output=str(output),
            )
        except Exception as e:
            return ToolResult(
                tool_call_id=tool_call.id,
                success=False,
                output="",
                error=str(e),
            )
```

### 3.4 Prompt 组装

```python
# agent/prompt_builder.py

class PromptBuilder:
    def build(self, role_def: RoleDefinition, shared_rules: List[SharedRule],
              assignment: NodeAssignment) -> str:
        """三层 prompt 组装"""

        parts = []

        # 第一层：共享规则（content 字段）
        for rule in shared_rules:
            if rule.enabled:
                parts.append(rule.content)

        # 第二层：角色身份 + 约束
        parts.append(f"# 角色身份\n{role_def.identity}")
        parts.append("# 行为约束\n" + "\n".join(f"- {c}" for c in role_def.constraints))

        # 第三层：当前任务上下文（来自 NodeAssignment）
        if assignment.action:
            parts.append(f"# 当前任务\n{assignment.action}")

        return "\n\n".join(parts)
```

### 3.6 角色加载（三级覆盖）

```python
# agent/role_loader.py

class RoleLoader:
    def load(self, role_id: str, project_id: Optional[str] = None) -> RoleDefinition:
        """从外部加载角色定义（Phase 1 本地文件，Phase 2 platform API）"""

        if self.config.load_from == "local":
            return self._load_from_local(role_id, project_id)
        else:
            return self._load_from_platform(role_id, project_id)

    def _load_from_local(self, role_id: str, project_id: Optional[str]) -> RoleDefinition:
        """Phase 1：从 sync 拉取到本地的文件加载"""
        # 项目级（最高优先级）
        if project_id:
            project_path = f"project_files/{project_id}/agents/role_store/{role_id}.yaml"
            if os.path.exists(project_path):
                return self._load_yaml(project_path)

        # 系统级（由 sync 从 platform 拉取到本地）
        system_path = f"project_files/_system/agents/role_store/{role_id}.yaml"
        return self._load_yaml(system_path)

    async def _load_from_platform(self, role_id: str, project_id: Optional[str]) -> RoleDefinition:
        """Phase 2：从 platform API 加载（自动三级覆盖）"""
        # platform 的 /api/agent-config/roles/{id} 接口自动处理三级覆盖
        params = {"project_id": project_id} if project_id else {}
        response = await self.platform_client.get(f"/api/agent-config/roles/{role_id}", params=params)
        return RoleDefinition(**response.json())
```

### 关键发现

在画流程图时发现了两个设计问题：

**问题 1：input_refs 的处理**

`input_refs: ["TASK-003"]` — runtime 需要在注入 prompt 之前，先从 platform 读取 TASK-003 的内容。这意味着 Dispatcher 需要一个 `resolve_input_refs` 步骤：

```python
# engine/dispatcher.py

async def resolve_input_refs(self, input_refs: List[str]) -> str:
    """将 input_refs 解析为自然语言文本，注入 prompt"""
    resolved = []
    for ref in input_refs:
        if ref.startswith("TASK-") or ref.startswith("REQ-") or ref.startswith("BUG-"):
            # 从 platform 读取工单内容
            workitem = await self.platform_client.get(f"/api/workitems/{ref}")
            resolved.append(f"## 工单 {ref}\n标题：{workitem['title']}\n描述：{workitem['description']}")
        elif ref.endswith(".output"):
            # 上一个节点的 output（从 flow 传递过来，存在 assignment 的 metadata 里）
            resolved.append(f"## 上一步产出\n{ref}")
        else:
            # 产物引用
            artifact = await self.platform_client.get(f"/api/artifacts/{ref}")
            resolved.append(f"## 产物 {ref}\n{artifact['name']}\n{artifact.get('structured_data', '')}")
    return "\n\n".join(resolved)
```

**问题 2：角色权限的 write_paths 需要包含 tests/**

后端开发的 `write_paths` 只有 `workspace/src/backend/`，但 TDD 流程需要写测试文件到 `workspace/tests/`。角色定义需要调整：

```yaml
# 后端开发角色定义
permissions:
  write_paths:
    - "workspace/src/backend/"
    - "workspace/tests/backend/"    # ← 需要加上测试目录
```

这两个问题需要在 platform 的初始数据（角色定义）中修正。

## 二、数据结构

### 2.1 SandboxInstance

```rust
// src/models.rs
//
// Sandbox 数据模型 — 项目沙箱环境的核心数据结构。
//
// 设计意图：
//   每个项目有一个 SandboxInstance，管理该项目的所有容器（Agent 容器 + 共享服务）。
//   三种模式（Dev/Preview/Build）对应不同的容器编排策略。

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SandboxInstance {
    pub id: String,                       // 沙箱唯一标识（如 "sandbox-blog-001"）
    pub project_id: String,               // 所属项目 ID
    pub mode: SandboxMode,                // 当前运行模式（决定容器编排方式）
    pub status: SandboxStatus,            // 当前生命周期状态
    pub tech_stack: TechStack,            // 技术栈配置（从 platform 读取，决定用什么镜像）
    pub containers: Vec<ContainerInfo>,   // 当前运行的容器列表（Agent + 共享服务）
    pub ports: PortMapping,               // 端口映射（项目级分配，避免多项目冲突）
    pub created_at: DateTime<Utc>,
    pub updated_at: DateTime<Utc>,
}

/// 沙箱运行模式 — 三种模式对应不同的容器编排策略
///
/// Dev：每个 Agent 一个独立容器 + 共享服务（数据库/Redis）
///   Agent 在各自容器中独立编码和测试，互不干扰。
/// Preview：停止 Agent 容器，启动统一部署容器（Nginx + 后端 + 前端）
///   用户可以像使用正式产品一样体验应用。
/// Build：启动构建容器，拉取最新代码，编译打包，上传产物
///   产出可部署的应用包（Docker 镜像 / 静态文件 / Web 应用包）。
#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum SandboxMode {
    Dev,        // 开发模式 — Agent 独立容器，实时编码
    Preview,    // 预览模式 — 统一部署，用户体验
    Build,      // 构建模式 — 编译打包，产出可部署产物
}

/// 沙箱状态 — 生命周期状态机
///
/// Creating → Running → Paused → Running（恢复）
/// Creating → Failed（创建失败：镜像拉取失败 / 端口冲突 / 磁盘满）
/// Running → Stopped（手动停止或项目完成）
#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum SandboxStatus {
    Creating,   // 正在创建（docker-compose up 中，等待健康检查）
    Running,    // 运行中（所有容器健康）
    Paused,     // 暂停（容器 pause，不释放资源，可快速恢复）
    Stopped,    // 已停止（容器 stop，释放 CPU/内存但保留磁盘卷）
    Failed,     // 创建失败（需要人工介入或重试）
}
```

### 2.2 TechStack（从 platform 读取）

```rust
// 技术栈配置 — 决定沙箱使用什么镜像和服务。
// 由 TL 在项目创建时指定，存储在 platform，sandbox 创建时读取。
//
// 为什么由 TL 指定而非自动检测：
//   Phase 1 项目从零开始（没有已有代码可检测），TL 在需求分析阶段决定技术栈。
//   Phase 2 可以从已有仓库的 Cargo.toml / package.json 自动推断。

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TechStack {
    pub backend: Option<StackConfig>,         // 后端配置（None = 无后端，如纯前端项目）
    pub frontend: Option<StackConfig>,        // 前端配置（None = 无前端，如纯 API 项目）
    pub database: Option<DatabaseConfig>,     // 数据库配置（None = 无数据库）
    pub extra_services: Vec<ServiceConfig>,   // 额外服务（Redis / Elasticsearch / RabbitMQ 等）
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct StackConfig {
    pub language: String,             // 编程语言："rust" | "python" | "node" | "go" | "java"
    pub framework: Option<String>,    // 框架："axum" | "fastapi" | "vue3" | "react" | "next"
    pub version: Option<String>,      // 语言版本："1.75" | "3.12" | "20"（用于选择 Docker 镜像 tag）
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct DatabaseConfig {
    pub db_type: String,              // 数据库类型："postgresql" | "mysql" | "mongodb" | "sqlite"
    pub version: String,              // 版本："16" | "8.0" | "7"（用于选择 Docker 镜像 tag）
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ServiceConfig {
    pub name: String,                 // 服务名："redis" | "elasticsearch" | "rabbitmq"
    pub image: String,                // Docker 镜像："redis:7-alpine" | "elasticsearch:8"
    pub ports: Vec<String>,           // 端口映射：["6379:6379"]（compose 格式）
}
```

### 2.3 ContainerInfo

```rust
/// 容器信息 — 沙箱中每个运行容器的状态描述
///
/// 一个 SandboxInstance 包含多个容器：
///   Dev 模式：N 个 Agent 容器 + M 个共享服务容器
///   Preview 模式：1 个 preview 容器 + M 个共享服务容器
///   Build 模式：1 个 builder 容器
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ContainerInfo {
    pub container_id: String,         // Docker container ID（短 hash）
    pub name: String,                 // 容器名（如 "sandbox-blog-be1"）— compose service name
    pub role: ContainerRole,          // 容器角色（决定生命周期管理方式）
    pub agent_id: Option<String>,     // dev 模式下关联的 Agent ID（共享服务为 None）
    pub image: String,                // 使用的 Docker 镜像（如 "rust:latest"）
    pub status: String,               // Docker 状态："running" | "exited" | "paused"
    pub ports: Vec<PortMap>,          // 端口映射列表
    pub created_at: DateTime<Utc>,
}

/// 容器角色 — 决定容器的生命周期和模式切换行为
///
/// Agent 容器：Dev 模式独有，切换到 Preview/Build 时停止
/// SharedService：所有模式共享（数据库/Redis），模式切换时保持运行
/// Preview：Preview 模式独有，切换回 Dev 时停止
/// Builder：Build 模式独有，构建完成后自动销毁
#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum ContainerRole {
    Agent,                            // dev 模式：Agent 独立容器（sleep infinity 等待 exec）
    SharedService,                    // 共享服务（数据库 / Redis）— 所有模式保持运行
    Preview,                          // preview 模式：统一部署容器（Nginx + 后端 + 前端）
    Builder,                          // build 模式：构建容器（构建完成后自动销毁）
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PortMap {
    pub host_port: u16,               // 宿主机端口
    pub container_port: u16,          // 容器内端口
    pub protocol: String,             // "tcp" | "udp"
}
```

### 2.4 PortMapping

```rust
/// 端口映射 — 项目级端口分配，避免多项目端口冲突
///
/// 每个项目分配一组连续端口（base_port + offset）：
///   base_port + 1 = backend
///   base_port + 2 = frontend
///   base_port + 3 = db
///   base_port + 4 = redis
///   base_port + 5 = preview
///   base_port + 6..9 = 预留
///
/// allocations 的 key 是服务名（"backend" / "frontend" / "db"），
/// value 是宿主机端口号。容器内端口固定（如 PostgreSQL 始终是 5432）。
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PortMapping {
    pub base_port: u16,               // 项目基础端口（如 4000）— 由 PortAllocator 分配
    pub allocations: HashMap<String, u16>,  // 服务名 → 宿主机端口
}
```

### 2.5 ExecRequest / ExecResult

```rust
/// Agent 工具调用 sandbox 执行命令 — runtime 的 code_execute 工具最终调用此接口
///
/// 数据流：Agent think→act → code_execute 工具 → SandboxProvider.exec()
///   → HTTP POST /sandbox/:id/exec → sandbox-manager → docker exec → 容器内执行
#[derive(Debug, Serialize, Deserialize)]
pub struct ExecRequest {
    pub agent_id: String,             // 在哪个 Agent 的容器中执行（查找容器的 key）
    pub command: String,              // shell 命令（如 "cargo test"）— 通过 /bin/sh -c 执行
    pub working_dir: Option<String>,  // 工作目录，默认 /workspace（容器内的项目根目录）
    pub timeout_secs: Option<u64>,    // 超时秒数，默认 300（5 分钟）— 防止死循环命令
    pub env: HashMap<String, String>, // 额外环境变量（如 RUST_LOG=debug）— 追加到容器默认环境
}

/// 命令执行结果 — 返回给 runtime，最终作为 ToolResult 返回给 LLM
#[derive(Debug, Serialize, Deserialize)]
pub struct ExecResult {
    pub exit_code: i32,               // 命令退出码 — 0=成功，非0=失败
    pub stdout: String,               // 标准输出 — 返回给 LLM 阅读
    pub stderr: String,               // 标准错误 — 失败时包含错误信息
    pub duration_ms: u64,             // 执行耗时（毫秒）— 用于 ExecutionTrace 记录
}
```

### 2.6 FileRequest

```rust
/// 写文件请求 — Agent 的 file_write 工具最终调用此接口
///
/// 写入策略（按文件大小自动选择）：
///   < 64KB：base64 编码后通过 docker exec echo 写入（快，无需额外 IO）
///   >= 64KB：通过 docker cp（tar 流）写入（支持大文件，避免命令行长度限制）
#[derive(Debug, Serialize, Deserialize)]
pub struct WriteFileRequest {
    pub agent_id: String,             // 目标 Agent 的容器
    pub path: String,                 // 相对于 /workspace 的路径（如 "src/models/tag.rs"）
    pub content: String,              // 文件内容
    pub create_dirs: bool,            // 是否自动创建父目录（mkdir -p）
}

/// 读文件请求 — Agent 的 file_read 工具最终调用此接口
#[derive(Debug, Serialize, Deserialize)]
pub struct ReadFileRequest {
    pub agent_id: String,             // 目标 Agent 的容器
    pub path: String,                 // 相对于 /workspace 的路径
}

/// 读文件结果
#[derive(Debug, Serialize, Deserialize)]
pub struct ReadFileResult {
    pub content: String,              // 文件内容（UTF-8 文本）
    pub size: u64,                    // 文件大小（字节）
    pub modified_at: DateTime<Utc>,   // 最后修改时间
}
```

### 3.2 compose_generator 实现

```rust
// docker/compose_generator.rs

pub struct ComposeGenerator {
    templates: TemplateRegistry,
}

impl ComposeGenerator {
    /// 根据项目配置生成 docker-compose.yml
    pub fn generate(&self, config: &CreateSandboxRequest) -> Result<String, SandboxError> {
        let mut compose = ComposeFile::new();

        // 1. 创建网络
        let network_name = format!("{}_net", config.project_id);
        compose.add_network(&network_name);

        // 2. 添加共享服务
        if let Some(db) = &config.tech_stack.database {
            let template = self.templates.get_db_template(&db.db_type, &db.version)?;
            compose.add_service("db", template.with_port(config.ports.get("db")));
        }
        for svc in &config.tech_stack.extra_services {
            compose.add_service(&svc.name, ServiceDef::from_config(svc));
        }

        // 3. 添加 Agent 容器
        for agent in &config.agents {
            let image = self.resolve_image(&agent.image_hint, &config.tech_stack)?;
            let svc = ServiceDef {
                image,
                working_dir: "/workspace".into(),
                command: vec!["sleep".into(), "infinity".into()],
                environment: self.build_env(&config.tech_stack),
                ports: self.agent_ports(agent, &config.ports),
                volumes: self.agent_volumes(agent, &config.project_id),
                networks: vec![network_name.clone()],
                depends_on: self.agent_depends_on(&config.tech_stack),
            };
            compose.add_service(&agent.agent_id, svc);
        }

        // 4. 序列化为 YAML
        Ok(compose.to_yaml()?)
    }

    /// 根据 image_hint 选择基础镜像
    fn resolve_image(&self, hint: &str, stack: &TechStack) -> Result<String, SandboxError> {
        match hint {
            "rust" => Ok(format!("rust:{}", stack.backend.as_ref()
                .and_then(|b| b.version.as_deref()).unwrap_or("latest"))),
            "node" => Ok(format!("node:{}", stack.frontend.as_ref()
                .and_then(|f| f.version.as_deref()).unwrap_or("20"))),
            "python" => Ok(format!("python:{}", stack.backend.as_ref()
                .and_then(|b| b.version.as_deref()).unwrap_or("3.12"))),
            "rust-node" => Ok("quantalithos/rust-node:latest".into()),
            _ => Err(SandboxError::UnsupportedImage(hint.into())),
        }
    }
}
```

### 4.1 exec 实现（最核心的 handler）

```rust
// api/exec.rs

pub async fn exec(
    State(state): State<AppState>,
    Path(sandbox_id): Path<String>,
    Json(req): Json<ExecRequest>,
) -> Result<Json<ExecResult>, SandboxError> {
    // 1. 查找 sandbox 实例
    let sandbox = state.instances.get(&sandbox_id)
        .ok_or(SandboxError::NotFound(sandbox_id.clone()))?;

    // 2. 查找 Agent 对应的容器
    let container = sandbox.containers.iter()
        .find(|c| c.agent_id.as_deref() == Some(&req.agent_id))
        .ok_or(SandboxError::AgentNotFound(req.agent_id.clone()))?;

    // 3. 在容器内执行命令
    let exec_opts = bollard::exec::CreateExecOptions {
        cmd: Some(vec!["/bin/sh", "-c", &req.command]),
        working_dir: Some(req.working_dir.as_deref().unwrap_or("/workspace")),
        env: Some(req.env.iter().map(|(k, v)| format!("{}={}", k, v)).collect()),
        attach_stdout: Some(true),
        attach_stderr: Some(true),
        ..Default::default()
    };

    let exec_id = state.docker.create_exec(&container.container_id, exec_opts).await?;
    let start = std::time::Instant::now();

    // 4. 启动执行（带超时）
    let timeout = Duration::from_secs(req.timeout_secs.unwrap_or(300));
    let output = tokio::time::timeout(
        timeout,
        state.docker.start_exec(&exec_id.id, None),
    ).await
        .map_err(|_| SandboxError::ExecTimeout(req.command.clone()))?
        .map_err(SandboxError::DockerError)?;

    // 5. 收集输出
    let (stdout, stderr) = collect_output(output).await;
    let inspect = state.docker.inspect_exec(&exec_id.id).await?;

    Ok(Json(ExecResult {
        exit_code: inspect.exit_code.unwrap_or(-1) as i32,
        stdout,
        stderr,
        duration_ms: start.elapsed().as_millis() as u64,
    }))
}
```

### 4.2 write_file 实现

```rust
// api/files.rs

pub async fn write_file(
    State(state): State<AppState>,
    Path(sandbox_id): Path<String>,
    Json(req): Json<WriteFileRequest>,
) -> Result<StatusCode, SandboxError> {
    let sandbox = state.instances.get(&sandbox_id)
        .ok_or(SandboxError::NotFound(sandbox_id.clone()))?;

    let container = sandbox.containers.iter()
        .find(|c| c.agent_id.as_deref() == Some(&req.agent_id))
        .ok_or(SandboxError::AgentNotFound(req.agent_id.clone()))?;

    // 1. 如果需要创建父目录
    if req.create_dirs {
        let dir = std::path::Path::new(&req.path).parent()
            .map(|p| p.to_string_lossy().to_string())
            .unwrap_or_default();
        if !dir.is_empty() {
            let mkdir_cmd = format!("mkdir -p /workspace/{}", dir);
            exec_in_container(&state.docker, &container.container_id, &mkdir_cmd).await?;
        }
    }

    // 2. 写入文件（通过 docker cp 或 exec echo）
    //    小文件用 exec，大文件用 docker cp
    let full_path = format!("/workspace/{}", req.path);
    if req.content.len() < 65536 {
        // 小文件：base64 编码后 exec 写入
        let encoded = base64::encode(&req.content);
        let cmd = format!("echo '{}' | base64 -d > {}", encoded, full_path);
        exec_in_container(&state.docker, &container.container_id, &cmd).await?;
    } else {
        // 大文件：docker cp
        let tar = create_tar_with_file(&req.path, req.content.as_bytes())?;
        state.docker.upload_to_container(
            &container.container_id,
            bollard::container::UploadToContainerOptions {
                path: "/workspace",
                ..Default::default()
            },
            tar.into(),
        ).await?;
    }

    Ok(StatusCode::CREATED)
}
```

```rust
pub async fn git_commit(sandbox_id: &str, agent_id: &str, message: &str) -> Result<ExecResult> {

    }).await
```

## 七、SandboxProvider 接口（core 定义）

```python
# core: sandbox/provider.py

class SandboxProvider(ABC):
    """沙箱提供者接口 — runtime 的工具通过此接口调用 sandbox"""

    @abstractmethod
    async def exec(self, sandbox_id: str, req: ExecRequest) -> ExecResult:
        """在指定 Agent 的容器内执行命令"""

    @abstractmethod
    async def write_file(self, sandbox_id: str, req: WriteFileRequest) -> None:
        """写文件到指定 Agent 的容器"""

    @abstractmethod
    async def read_file(self, sandbox_id: str, req: ReadFileRequest) -> ReadFileResult:
        """读取指定 Agent 容器内的文件"""

    @abstractmethod
    async def git_commit(self, sandbox_id: str, agent_id: str, message: str) -> ExecResult:
        """在 Agent 容器内 git commit"""

    @abstractmethod
    async def git_push(self, sandbox_id: str, agent_id: str) -> ExecResult:
        """在 Agent 容器内 git push"""

    @abstractmethod
    async def get_service_url(self, sandbox_id: str, service: str) -> str:
        """获取共享服务的 URL（如数据库连接串）"""


class HttpSandboxProvider(SandboxProvider):
    """通过 HTTP 调用 sandbox-manager API 的实现"""

    def __init__(self, base_url: str):
        self.base_url = base_url  # "http://sandbox-manager:8004"
        self.client = httpx.AsyncClient()

    async def exec(self, sandbox_id: str, req: ExecRequest) -> ExecResult:
        resp = await self.client.post(
            f"{self.base_url}/sandbox/{sandbox_id}/exec",
            json=req.dict(),
        )
        resp.raise_for_status()
        return ExecResult(**resp.json())

    async def write_file(self, sandbox_id: str, req: WriteFileRequest) -> None:
        resp = await self.client.post(
            f"{self.base_url}/sandbox/{sandbox_id}/write-file",
            json=req.dict(),
        )
        resp.raise_for_status()

    # ... 其他方法类似
```

```rust
// docker/port_allocator.rs

pub struct PortAllocator {
    base: u16,                        // 起始端口（如 4000）
    step: u16,                        // 每个项目占用的端口数（如 10）
    allocated: HashMap<String, u16>,  // project_id → base_port
}

impl PortAllocator {
    /// 为项目分配一组端口
    pub fn allocate(&mut self, project_id: &str) -> PortMapping {
        let base = self.next_available();
        self.allocated.insert(project_id.to_string(), base);

        PortMapping {
            base_port: base,
            allocations: HashMap::from([
                ("backend".into(), base + 1),
                ("frontend".into(), base + 2),
                ("db".into(), base + 3),
                ("redis".into(), base + 4),
                ("preview".into(), base + 5),
                // base + 6..9 预留
            ]),
        }
    }

    fn next_available(&self) -> u16 {
        let max = self.allocated.values().max().copied().unwrap_or(self.base);
        max + self.step
    }
}
```

## 九、设计决策：代码在容器内 vs 宿主机

```
方案 A：代码在容器内（采纳）

  Agent file_write → 容器内 /workspace/blog/src/tag.py
  Agent code_execute → 容器内 cargo test
  Agent git_push → 容器内 → Gitea
  你看代码 → git clone 从 Gitea 拉

  ┌──────────────────────────────────────────────┐
  │  sandbox 容器                                 │
  │  /workspace/blog/src/tag.py  ← 代码在这里     │
  │  .git/                       ← Git 在这里     │
  │       │                                      │
  │       │ git push（每次 commit 后自动）         │
  │       ▼                                      │
  │  Gitea blog-backend.git      ← 持久化备份     │
  └──────────────────────────────────────────────┘
       │
       │ 你在本地：git clone / git pull
       ▼
  ~/blog-backend/                ← 本地副本

  优点：
    ✅ 完全隔离，Agent 不能访问宿主机
    ✅ 容器销毁后代码不丢（Gitea 有备份）
    ✅ 环境一致性（容器内的 Rust 版本和运行时一致）
  缺点：
    ❌ 不能直接在本地 IDE 打开（需要 git clone 一份）


方案 B：代码在宿主机（未采纳）

  Agent file_write → 宿主机 ~/projects/blog/src/tag.py
  容器挂载 ~/projects/blog/ → 容器内编译运行

  优点：
    ✅ 直接在本地 IDE 打开
    ✅ 代码始终在宿主机
  缺点：
    ❌ Agent 能访问宿主机文件系统（安全风险）
    ❌ 多项目文件混在一起
    ❌ 环境不一致（宿主机和容器的依赖版本可能不同）


选择方案 A 的原因：
  安全性优先 — AI Agent 在沙箱中执行，不能访问宿主机
  通过 Gitea + git clone 解决"本地查看代码"的需求
  Phase 2 可以加 VS Code Remote 直接连接容器内编辑
```

## 十、错误处理

```rust
// src/error.rs
//
// 统一错误类型 — sandbox-manager 所有 handler 的错误返回格式。
//
// 可重试性标注：
//   NotFound — 不可重试（沙箱/Agent 不存在，需要先创建）
//   AgentNotFound — 不可重试（Agent 容器不存在）
//   DockerError — 视情况（网络问题可重试，镜像不存在不可重试）
//   ExecTimeout — 不可重试（命令本身太慢，重试也会超时）
//   UnsupportedImage — 不可重试（image_hint 无效，需要修正配置）
//   GitError — 视情况（网络问题可重试，冲突不可重试）
//   PortExhausted — 不可重试（需要释放其他项目的端口或扩大端口范围）
//   ModeSwitchFailed — 可重试（可能是容器暂时不可达）
//   BuildFailed — 不可重试（代码编译错误，需要修复代码）

#[derive(Debug, thiserror::Error)]
pub enum SandboxError {
    #[error("Sandbox not found: {0}")]
    NotFound(String),                 // 沙箱实例不存在

    #[error("Agent not found in sandbox: {0}")]
    AgentNotFound(String),            // 指定的 Agent 容器不存在

    #[error("Docker error: {0}")]
    DockerError(#[from] bollard::errors::Error),  // Docker API 调用失败

    #[error("Exec timeout: {0}")]
    ExecTimeout(String),              // 命令执行超时（超过 timeout_secs）

    #[error("Unsupported image: {0}")]
    UnsupportedImage(String),         // 不支持的 image_hint（resolve_image 失败）

    #[error("Git error: {0}")]
    GitError(String),                 // Git 操作失败（commit/push）

    #[error("Port allocation failed: no available ports")]
    PortExhausted,                    // 端口耗尽（所有端口段都被占用）

    #[error("Mode switch failed: {0}")]
    ModeSwitchFailed(String),         // 模式切换失败（如 preview 容器启动失败）

    #[error("Build failed: {0}")]
    BuildFailed(String),              // 构建失败（编译错误 / 测试失败）
}
```

## 十三、并发与一致性

### 13.1 竞态场景分析

| 竞态场景 | 共享资源 | 竞态类型 | 控制策略 | 一致性模型 |
|---------|---------|---------|---------|-----------|
| 同一项目并发 create | Docker daemon + 端口池 | 写-写 | sandbox_create 内部 tokio Mutex 串行化 | 强一致 |
| create 与 destroy 并发 | SandboxInstance 状态 | 写-写 | 按 sandbox_id 加锁（DashMap 分片锁） | 强一致 |
| 多 Agent 并发 exec 同一容器 | 容器内进程 | 写-写 | Docker exec 本身支持并发；sandbox-manager 不额外加锁 | 最终一致 |
| exec 与 destroy 并发 | 容器生命周期 | 读-写 | exec 前检查容器状态；destroy 等待进行中的 exec 完成（最长 10s） | 强一致 |
| 多 Agent 并发 write_file 同一文件 | 容器内文件 | 写-写 | sandbox-manager 不加锁（由 runtime 层保证同一文件不并发写） | 依赖上游 |
| 端口分配并发 | port_allocator 内部 HashSet | 写-写 | tokio Mutex 保护端口分配操作 | 强一致 |

### 13.2 事务边界

sandbox-manager 无数据库，状态保存在内存中（HashMap<SandboxId, SandboxInstance>）。

```
状态一致性保证：

  create:
    1. 分配端口（Mutex 保护）
    2. 生成 docker-compose.yml
    3. docker-compose up
    4. 注册到内存 HashMap（DashMap）
    → 如果 step 3 失败：释放端口，不注册到 HashMap

  destroy:
    1. 从 HashMap 标记为 destroying（DashMap 写锁）
    2. docker-compose down
    3. 释放端口
    4. 从 HashMap 移除
    → 如果 step 2 失败：保留 destroying 状态，日志告警，人工处理
```

### 13.3 声明模型之外的例外

- 容器内文件系统的并发由 Linux 内核保证（POSIX 语义），sandbox-manager 不额外管理
- Docker daemon 自身的并发控制由 Docker Engine 保证，bollard 客户端是线程安全的
- port_allocator 在进程重启后通过 `docker ps` 扫描已占用端口重建状态，可能存在短暂的端口冲突窗口（< 1s）

## 十四、性能设计

### 14.1 容量估算

| 指标 | Phase 1 | Phase 2 | 依据 |
|------|---------|---------|------|
| 并发项目数 | 3 | 20 | 单机 Docker 资源限制 |
| 单项目容器数 | 2–5 | 5–10 | Agent 数 + 共享服务 |
| sandbox_create 延迟 | < 30s | < 15s | 镜像缓存命中 + compose up |
| exec 延迟（简单命令） | < 2s | < 2s | Docker exec API 开销 |
| write_file 延迟 | < 500ms | < 500ms | Docker cp 或 exec echo |
| sandbox-manager 内存 | < 128MB | < 256MB | 主要消耗：HTTP server + 状态 HashMap |

### 14.2 热点优化

| 热点 | 优化方案 |
|------|---------|
| sandbox_create（最慢操作） | 预构建基础镜像（避免每次 pull）；compose up 并行启动容器 |
| exec（最频繁操作） | bollard 连接池复用；不为每次 exec 创建新连接 |
| 镜像拉取 | 本地镜像缓存；Phase 2 引入私有 registry |
| 端口分配 | 内存 HashSet O(1) 查找；范围 10000–60000 足够 50 个项目 |

### 14.3 资源限制（Phase 2）

| 资源 | 限制方式 | 默认值 |
|------|---------|--------|
| 内存 | Docker --memory | 2GB/容器 |
| CPU | Docker --cpus | 2 核/容器 |
| 磁盘 | Docker --storage-opt size | 10GB/容器（Phase 3） |
| 网络 | Docker network isolation | 项目间隔离，容器间同网络 |

## 十八、跨模块调用链

### 18.1 Agent 执行代码（runtime → sandbox → Docker）

```
flow 发送 FlowMessage(NodeExecutePayload) 给 runtime
  → runtime AgentWorkflowEngine 执行到 code_execute 节点
    → runtime 调用 SandboxProvider.exec(sandbox_id, container_id, command)
      → HttpSandboxProvider POST http://sandbox-manager:8004/sandbox/sb-001/exec
        Body: { "container_id": "blog-be-1", "command": ["cargo", "build"], "timeout_secs": 60 }
        → sandbox-manager api::exec::exec_command
          → state.sandbox_store.get("sb-001") → 检查状态 = running
          → container_manager.exec("blog-be-1", ["cargo", "build"], 60s)
            → bollard docker.create_exec("blog-be-1", CreateExecOptions { cmd, ... })
            → bollard docker.start_exec(exec_id)
            → 读取 stdout/stderr 流，等待退出
          → 返回 ExecResult { exit_code: 0, stdout: "...", stderr: "" }
        → HTTP 200 + JSON
      → HttpSandboxProvider 反序列化为 ExecResult
    → runtime 将结果写入 NodeResult
  → runtime 提交 NodeResult 给 flow
```

### 18.2 Agent 写文件（runtime → sandbox → Docker）

```
runtime 调用 SandboxProvider.write_file(sandbox_id, container_id, path, content)
  → HttpSandboxProvider POST http://sandbox-manager:8004/sandbox/sb-001/write-file
    Body: { "container_id": "blog-be-1", "path": "/workspace/blog-backend/src/main.rs", "content": "..." }
    → sandbox-manager api::files::write_file
      → 路径校验：path 必须在 /workspace/ 下，不允许 .. 穿越
      → container_manager.exec("blog-be-1",
          ["sh", "-c", "cat > /workspace/blog-backend/src/main.rs"],
          stdin: content)
      → 返回 200 OK
```

### 18.3 项目创建完整链路（platform → sandbox → Docker + Gitea）

```
用户在 console 创建项目 "blog"
  → console → gate → platform POST /api/projects { name: "blog", tech_stack: [...] }
    → platform 创建项目记录
    → platform 发布 ProjectCreated 事件
      → flow 监听到事件，启动 Pipeline 流程
        → flow 阶段 0：调用 sandbox
          → runtime 调用 SandboxProvider.create(project_id, mode=dev, tech_stack)
            → sandbox-manager POST /sandbox/create
              → 读取 tech_stack → 选择基础镜像
              → compose_generator 生成 docker-compose.yml
              → port_allocator 分配端口
              → docker-compose up -d
              → Phase 2: gitea_client.create_repo("blog-backend") + create_repo("blog-frontend")
              → Phase 2: 容器内 git clone
              → 注册到 sandbox_store
              → 返回 sandbox_id + 容器列表 + 端口映射
```

### 18.4 preview 模式切换（Phase 2）

```
TL Agent 完成阶段 4 验收
  → flow 触发 preview 模式

        → 分配 preview 端口
        → 通知 gate 注册反向代理规则
        → 返回 preview_url: "https://preview-blog.quantalithos.local"
```

### 18.5 构建产物上传（Phase 2）

```
TL Agent 触发 build 模式
  → runtime 调用 SandboxProvider.switch_mode(sandbox_id, "build")
    → sandbox-manager POST /sandbox/sb-001/switch { mode: "build" }
      → 启动 build 容器
      → 容器内：git pull + cargo build --release + npm run build + 集成测试
      → 打包产物（Docker 镜像 / tar.gz / Web 应用包）
      → sandbox-manager 调用 platform POST /api/projects/blog/artifacts
        Body: { type: "build_output", files: [...], test_result: "passed" }
      → platform 存储产物元数据

```

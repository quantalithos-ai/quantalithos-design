## 二、数据结构

### 2.1 Device（设备）

```rust
// device/registry.rs
//
// 设备数据模型 — 描述一台可用于测试的物理设备或模拟器。
//
// 设计意图：
//   testbed 管理一组测试设备（真机 + 模拟器），
//   Tester Agent 通过 device_list 工具查看可用设备，
//   通过 device_deploy / device_run_test 在指定设备上执行测试。
//
// connection 字段决定如何连接设备：
//   usb — 通过 USB 连接（iOS 真机必须 USB 连接到 macOS）
//   adb_wifi — Android ADB over WiFi（设备和 testbed 在同一网络）
//   adb_usb — Android ADB over USB
//   ssh — SSH 远程连接（macOS/Windows/Linux 桌面设备）
//   emulator — 本地模拟器（testbed 同机运行的 Android Emulator / iOS Simulator）
//   cloud — 云设备农场（Phase 4，通过 adapter 对接）

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Device {
    pub id: String,                      // 设备唯一标识（如 "iphone-14-lab1"）
    pub platform: Platform,              // 平台类型
    pub model: String,                   // 设备型号（如 "iPhone 14 Pro" / "Pixel 7"）
    pub os_version: String,              // 系统版本（如 "iOS 17.2" / "Android 14"）
    pub connection: ConnectionConfig,    // 连接配置
    pub status: DeviceStatus,            // 当前状态
    pub capabilities: Vec<String>,       // 设备能力（"camera" / "gps" / "nfc" / "biometric"）
    pub current_job_id: Option<String>,  // 当前正在执行的任务（busy 时有值）
    pub last_heartbeat: DateTime<Utc>,   // 最后一次健康检查时间
    pub registered_at: DateTime<Utc>,    // 注册时间
}

/// 平台类型
#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum Platform {
    Ios,          // iPhone / iPad
    Android,      // Android 手机 / 平板
    Macos,        // macOS 桌面
    Windows,      // Windows 桌面
    Linux,        // Linux 桌面
}

/// 设备状态
#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum DeviceStatus {
    Available,    // 空闲，可接受新任务
    Busy,         // 正在执行测试任务
    Offline,      // 离线（健康检查失败）
    Maintenance,  // 维护中（手动标记，不接受任务）
}

/// 连接配置 — 不同连接方式有不同的参数
#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(tag = "type")]
pub enum ConnectionConfig {
    #[serde(rename = "usb")]
    Usb {
        udid: String,                    // iOS 设备 UDID / Android serial
        host: Option<String>,            // 设备连接到哪台机器（如 macOS 构建机 IP）
    },
    #[serde(rename = "adb_wifi")]
    AdbWifi {
        ip: String,                      // 设备 IP
        port: u16,                       // ADB 端口（默认 5555）
    },
    #[serde(rename = "ssh")]
    Ssh {
        host: String,                    // SSH 主机地址
        port: u16,                       // SSH 端口（默认 22）
        user: String,                    // SSH 用户名
        key_path: Option<String>,        // SSH 私钥路径（None = 密码认证）
    },
    #[serde(rename = "emulator")]
    Emulator {
        emulator_name: String,           // 模拟器名称（如 "Pixel_7_API_34"）
        avd_path: Option<String>,        // AVD 路径（Android）
    },
    #[serde(rename = "cloud")]
    Cloud {
        provider: String,                // "aws" | "firebase" | "browserstack"
        device_arn: String,              // 云设备标识
    },
}
```

### 2.2 TestJob（测试任务）

```rust
// job/executor.rs
//
// 测试任务 — 一次完整的"部署 + 测试 + 收集"流程。
//
// 生命周期：
//   queued → deploying → running → collecting → completed/failed
//
// 为什么有 queued 状态：
//   多个 Agent 可能同时请求同一台设备，需要排队。
//   scheduler 按 FIFO + 优先级调度。

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TestJob {
    pub job_id: String,                  // 任务唯一标识（UUID）
    pub device_id: String,               // 目标设备 ID
    pub artifact_id: String,             // 要部署的应用（platform artifact ID）
    pub artifact_url: String,            // 应用下载地址
    pub test_script: TestScript,         // 测试脚本（解析后的结构）
    pub status: JobStatus,               // 当前状态
    pub priority: i32,                   // 优先级（数字越大越优先，默认 0）
    pub requested_by: String,            // 请求者 Agent ID
    pub project_id: String,              // 所属项目
    pub created_at: DateTime<Utc>,       // 创建时间
    pub started_at: Option<DateTime<Utc>>,   // 开始执行时间
    pub completed_at: Option<DateTime<Utc>>, // 完成时间
    pub result: Option<TestResult>,      // 执行结果（completed/failed 时有值）
}

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum JobStatus {
    Queued,       // 排队中（等待设备空闲）
    Deploying,    // 正在部署应用到设备
    Running,      // 正在执行测试脚本
    Collecting,   // 正在收集结果（截图/日志/性能）
    Completed,    // 执行完成（不代表测试通过，看 result.status）
    Failed,       // 执行失败（部署失败/设备断开/超时）
    Cancelled,    // 被取消
}
```

### 2.3 TestScript（测试脚本）

```rust
// job/script_parser.rs
//
// 测试脚本 — 声明式 YAML 格式，由 Tester Agent 编写。
//
// 设计意图：
//   用声明式 YAML 而非命令式代码（如 Python + Appium），原因：
//   1. Agent 更容易生成 YAML（结构化输出）
//   2. 人类更容易审阅（一眼看出测试步骤）
//   3. 跨平台统一（同一脚本在 iOS/Android 上都能跑）
//   4. 安全（不能执行任意代码，只能执行预定义的 action）
//
// 元素定位策略（优先级）：
//   accessibility_id — 最推荐（跨平台一致，Flutter/RN 原生支持）
//   text — 按显示文本定位（简单但不稳定，文案变了就失败）
//   xpath — 最后手段（复杂但精确）

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TestScript {
    pub name: String,                    // 脚本名称（如 "camera_capture_test"）
    pub timeout: Duration,               // 整体超时（默认 120s）
    pub setup: Option<SetupConfig>,      // 前置配置（如清除应用数据）
    pub steps: Vec<TestStep>,            // 测试步骤列表
    pub teardown: Option<TeardownConfig>, // 后置清理（如卸载应用）
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SetupConfig {
    pub clear_app_data: bool,            // 执行前清除应用数据
    pub grant_permissions: Vec<String>,  // 预授权权限（跳过弹窗）
    pub install_dependencies: Vec<String>, // 前置安装（如测试数据包）
}

/// 测试步骤 — 每一步是一个原子操作
#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(tag = "action")]
pub enum TestStep {
    /// 启动应用
    #[serde(rename = "launch_app")]
    LaunchApp {
        package: String,                 // 包名（如 "com.quantalithos.diary"）
    },

    /// 点击元素
    #[serde(rename = "tap")]
    Tap {
        element: ElementLocator,         // 元素定位
        timeout: Option<Duration>,       // 等待元素出现的超时
    },

    /// 输入文本
    #[serde(rename = "type_text")]
    TypeText {
        element: ElementLocator,
        text: String,
    },

    /// 滑动
    #[serde(rename = "swipe")]
    Swipe {
        direction: String,               // up / down / left / right
        distance: Option<f32>,           // 滑动距离（屏幕百分比，0.0-1.0）
    },

    /// 等待
    #[serde(rename = "wait")]
    Wait {
        duration: Duration,
    },

    /// 断言元素可见
    #[serde(rename = "assert_visible")]
    AssertVisible {
        element: ElementLocator,
        screenshot: Option<String>,      // 截图保存名（如 "step_permission"）
        timeout: Option<Duration>,
    },

    /// 断言元素不可见
    #[serde(rename = "assert_not_visible")]
    AssertNotVisible {
        element: ElementLocator,
    },

    /// 断言文本内容
    #[serde(rename = "assert_text")]
    AssertText {
        element: ElementLocator,
        expected: String,
    },

    /// 截图
    #[serde(rename = "screenshot")]
    Screenshot {
        name: String,                    // 截图名称
    },

    /// 检查性能
    #[serde(rename = "check_performance")]
    CheckPerformance {
        metrics: Vec<String>,            // ["memory", "cpu", "fps"]
        threshold: PerformanceThreshold,
    },

    /// 检查无崩溃
    #[serde(rename = "check_no_crash")]
    CheckNoCrash {
        duration: Duration,              // 观察时长
    },

    /// 返回/后退
    #[serde(rename = "back")]
    Back,
}

/// 元素定位器 — 支持多种定位策略
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ElementLocator {
    pub accessibility_id: Option<String>,  // 无障碍 ID（最推荐）
    pub text: Option<String>,              // 显示文本
    pub xpath: Option<String>,             // XPath 表达式
    pub id: Option<String>,                // 平台原生 ID（Android resource-id / iOS identifier）
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PerformanceThreshold {
    pub memory_mb: Option<u64>,          // 内存上限（MB）
    pub cpu_percent: Option<u32>,        // CPU 使用率上限
    pub fps_min: Option<u32>,            // 最低帧率
}
```

### 2.4 TestResult（测试结果）

```rust
// result/collector.rs
//
// 测试结果 — 一次测试任务的完整执行报告。
//
// 上报到 platform 后作为 Artifact 存储（type = "device_test_report"）。
// Tester Agent 读取结果后决定：通过 → 继续 / 失败 → 创建 BUG 工单。

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TestResult {
    pub job_id: String,                  // 关联的测试任务
    pub device_id: String,               // 执行设备
    pub device_model: String,            // 设备型号（方便阅读）
    pub platform: Platform,              // 平台
    pub os_version: String,              // 系统版本
    pub status: TestStatus,              // 总体结果
    pub steps_total: u32,                // 总步骤数
    pub steps_passed: u32,               // 通过步骤数
    pub steps_failed: u32,               // 失败步骤数
    pub step_results: Vec<StepResult>,   // 每步的详细结果
    pub screenshots: Vec<Screenshot>,    // 截图列表
    pub device_logs: String,             // 设备日志（测试期间）
    pub crash_report: Option<String>,    // 崩溃报告（如果有）
    pub performance: PerformanceReport,  // 性能数据
    pub duration_ms: u64,                // 总执行时长
    pub completed_at: DateTime<Utc>,     // 完成时间
}

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum TestStatus {
    Passed,       // 所有步骤通过
    Failed,       // 有步骤失败（断言不满足）
    Error,        // 执行错误（设备断开/Appium 异常）
    Timeout,      // 整体超时
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct StepResult {
    pub step_index: u32,                 // 步骤序号（从 0 开始）
    pub action: String,                  // 动作类型（如 "tap" / "assert_visible"）
    pub status: StepStatus,              // passed / failed / skipped / error
    pub message: Option<String>,         // 失败/错误时的详细信息
    pub duration_ms: u64,                // 该步耗时
    pub screenshot: Option<String>,      // 该步截图文件名
}

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
pub enum StepStatus {
    Passed,
    Failed,       // 断言失败（元素不存在/文本不匹配）
    Skipped,      // 跳过（前置步骤失败后的后续步骤）
    Error,        // 执行错误（Appium 异常）
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Screenshot {
    pub name: String,                    // 截图名称（如 "step_permission"）
    pub step_index: u32,                 // 关联的步骤序号
    pub file_path: String,              // 本地文件路径
    pub width: u32,
    pub height: u32,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PerformanceReport {
    pub memory_peak_mb: u64,             // 内存峰值
    pub memory_avg_mb: u64,              // 内存平均值
    pub cpu_peak_percent: u32,           // CPU 峰值
    pub cpu_avg_percent: u32,            // CPU 平均值
    pub fps_min: Option<u32>,            // 最低帧率（如果能采集）
    pub fps_avg: Option<u32>,            // 平均帧率
    pub battery_drain_percent: Option<f32>, // 电量消耗百分比
}
```

### 3.2 设备排队调度

```
场景：两个 Agent 同时请求 pixel-7

  10:00  Tester-1 请求 pixel-7 → 设备 available → 立即执行，标记 busy
  10:01  Tester-2 请求 pixel-7 → 设备 busy → 加入队列（queued）
  10:05  Tester-1 的任务完成 → 设备 available → 从队列取出 Tester-2 的任务 → 执行

调度策略：
  1. FIFO（先到先得）— 默认
  2. 优先级（priority 字段）— 高优先级插队
  3. 超时保护 — 排队超过 5 分钟自动失败（防止无限等待）
```

### 3.3 健康检查

```rust
// device/health.rs
//
// 设备健康检查 — 定期 ping 所有注册设备，离线自动标记。
//
// 检查间隔：30 秒
// 离线判定：连续 3 次 ping 失败 → 标记 offline
// 恢复判定：ping 成功 → 标记 available（如果之前是 offline）

pub struct HealthChecker {
    registry: Arc<DeviceRegistry>,
    check_interval: Duration,            // 30s
    max_failures: u32,                   // 3 次失败 → offline
}

impl HealthChecker {
    /// 启动后台健康检查循环
    pub async fn start(&self) {
        loop {
            for device in self.registry.list_all().await {
                let reachable = self.ping(&device).await;
                if !reachable {
                    let failures = self.registry.increment_failure(&device.id).await;
                    if failures >= self.max_failures {
                        self.registry.set_status(&device.id, DeviceStatus::Offline).await;
                        tracing::warn!(device_id = %device.id, "device went offline");
                    }
                } else {
                    self.registry.reset_failure(&device.id).await;
                    if device.status == DeviceStatus::Offline {
                        self.registry.set_status(&device.id, DeviceStatus::Available).await;
                        tracing::info!(device_id = %device.id, "device back online");
                    }
                }
            }
            tokio::time::sleep(self.check_interval).await;
        }
    }

    /// ping 设备 — 根据连接方式选择检查方法
    async fn ping(&self, device: &Device) -> bool {
        match &device.connection {
            ConnectionConfig::AdbWifi { ip, port } => {
                // adb connect {ip}:{port} && adb -s {ip}:{port} shell echo ok
                self.adb_ping(ip, *port).await
            },
            ConnectionConfig::Ssh { host, port, user, .. } => {
                // ssh -o ConnectTimeout=5 {user}@{host} echo ok
                self.ssh_ping(host, *port, user).await
            },
            ConnectionConfig::Usb { udid, host } => {
                // ideviceinfo -u {udid} 或 adb -s {udid} shell echo ok
                self.usb_ping(udid, host.as_deref()).await
            },
            ConnectionConfig::Emulator { emulator_name, .. } => {
                // adb -s emulator-5554 shell echo ok
                self.emulator_ping(emulator_name).await
            },
            ConnectionConfig::Cloud { .. } => {
                // 云设备始终 available（由云服务管理）
                true
            },
        }
    }
}
```

```rust
// driver/android.rs
//
// Android 驱动 — ADB 部署 + Appium UiAutomator2 测试。
//
// 部署方式：adb -s {serial} install -r {apk_path}
// 测试方式：Appium HTTP API（W3C WebDriver 协议）
// 日志获取：adb logcat -d -t {timestamp}
// 性能获取：adb shell dumpsys meminfo {package} + adb shell top -n 1

pub struct AndroidDriver {
    appium_url: String,                  // Appium server 地址（如 "http://localhost:4723"）
    adb_path: String,                    // ADB 可执行文件路径
}

#[async_trait]
impl DeviceDriver for AndroidDriver {
    async fn deploy(&self, device: &Device, artifact_path: &str) -> Result<DeployResult, TestbedError> {
        let serial = self.get_serial(device)?;
        let start = Instant::now();

        // adb install（-r = 覆盖安装）
        let output = Command::new(&self.adb_path)
            .args(["-s", &serial, "install", "-r", artifact_path])
            .output()
            .await
            .map_err(|e| TestbedError::DeployFailed(e.to_string()))?;

        if !output.status.success() {
            return Err(TestbedError::DeployFailed(
                String::from_utf8_lossy(&output.stderr).to_string()
            ));
        }

        Ok(DeployResult {
            success: true,
            install_time_ms: start.elapsed().as_millis() as u64,
        })
    }

    async fn create_session(&self, device: &Device, app_package: &str) -> Result<SessionId, TestbedError> {
        // 通过 Appium HTTP API 创建 session
        let capabilities = serde_json::json!({
            "platformName": "Android",
            "appium:automationName": "UiAutomator2",
            "appium:deviceName": device.model,
            "appium:udid": self.get_serial(device)?,
            "appium:appPackage": app_package,
            "appium:noReset": true,
        });

        let resp = reqwest::Client::new()
            .post(format!("{}/session", self.appium_url))
            .json(&serde_json::json!({ "capabilities": { "alwaysMatch": capabilities } }))
            .send().await?;

        let body: serde_json::Value = resp.json().await?;
        let session_id = body["value"]["sessionId"].as_str()
            .ok_or(TestbedError::SessionFailed("No session ID in response".into()))?;

        Ok(SessionId(session_id.to_string()))
    }

    async fn execute_step(&self, session: &SessionId, step: &TestStep) -> Result<StepResult, TestbedError> {
        let start = Instant::now();

        match step {
            TestStep::Tap { element, timeout } => {
                let el = self.find_element(session, element, *timeout).await?;
                self.click_element(session, &el).await?;
                Ok(StepResult { status: StepStatus::Passed, duration_ms: start.elapsed().as_millis() as u64, .. })
            },
            TestStep::AssertVisible { element, screenshot, timeout } => {
                match self.find_element(session, element, *timeout).await {
                    Ok(_) => {
                        if let Some(name) = screenshot {
                            let img = self.screenshot(session).await?;
                            // 保存截图
                        }
                        Ok(StepResult { status: StepStatus::Passed, .. })
                    },
                    Err(_) => Ok(StepResult {
                        status: StepStatus::Failed,
                        message: Some(format!("Element not visible: {:?}", element)),
                        ..
                    }),
                }
            },
            // ... 其他步骤类型
            _ => todo!(),
        }
    }

    // ... 其他方法
}
```

## 五、错误处理

```rust
// src/error.rs
//
// testbed 统一错误类型。
//
// 错误分类：
//   设备相关：DeviceNotFound / DeviceOffline / DeviceBusy
//   部署相关：DeployFailed / ArtifactNotFound / ArtifactDownloadFailed
//   测试相关：SessionFailed / StepFailed / ScriptParseError / TestTimeout
//   系统相关：InternalError

#[derive(Debug, thiserror::Error)]
pub enum TestbedError {
    #[error("Device not found: {0}")]
    DeviceNotFound(String),              // 设备 ID 不存在

    #[error("Device offline: {0}")]
    DeviceOffline(String),               // 设备离线（健康检查失败）

    #[error("Device busy: {0}")]
    DeviceBusy(String),                  // 设备正在执行其他任务

    #[error("Deploy failed: {0}")]
    DeployFailed(String),                // 应用部署失败（安装错误）

    #[error("Artifact not found: {0}")]
    ArtifactNotFound(String),            // platform 上找不到构建产物

    #[error("Artifact download failed: {0}")]
    ArtifactDownloadFailed(String),      // 下载构建产物失败

    #[error("Session failed: {0}")]
    SessionFailed(String),               // Appium session 创建失败

    #[error("Step execution failed: {0}")]
    StepFailed(String),                  // 测试步骤执行异常（非断言失败）

    #[error("Script parse error: {0}")]
    ScriptParseError(String),            // YAML 测试脚本格式错误

    #[error("Test timeout after {0}s")]
    TestTimeout(u64),                    // 测试整体超时

    #[error("Queue timeout: device {0} busy for too long")]
    QueueTimeout(String),                // 排队超时（设备长时间不空闲）

    #[error("Internal error: {0}")]
    InternalError(String),               // 未预期的内部错误
}

impl IntoResponse for TestbedError {
    fn into_response(self) -> Response {
        let (status, code) = match &self {
            Self::DeviceNotFound(_) => (StatusCode::NOT_FOUND, "DEVICE_NOT_FOUND"),
            Self::DeviceOffline(_) => (StatusCode::SERVICE_UNAVAILABLE, "DEVICE_OFFLINE"),
            Self::DeviceBusy(_) => (StatusCode::CONFLICT, "DEVICE_BUSY"),
            Self::DeployFailed(_) => (StatusCode::BAD_GATEWAY, "DEPLOY_FAILED"),
            Self::ArtifactNotFound(_) => (StatusCode::NOT_FOUND, "ARTIFACT_NOT_FOUND"),
            Self::ArtifactDownloadFailed(_) => (StatusCode::BAD_GATEWAY, "DOWNLOAD_FAILED"),
            Self::SessionFailed(_) => (StatusCode::BAD_GATEWAY, "SESSION_FAILED"),
            Self::StepFailed(_) => (StatusCode::INTERNAL_SERVER_ERROR, "STEP_FAILED"),
            Self::ScriptParseError(_) => (StatusCode::BAD_REQUEST, "SCRIPT_PARSE_ERROR"),
            Self::TestTimeout(_) => (StatusCode::GATEWAY_TIMEOUT, "TEST_TIMEOUT"),
            Self::QueueTimeout(_) => (StatusCode::GATEWAY_TIMEOUT, "QUEUE_TIMEOUT"),
            Self::InternalError(_) => (StatusCode::INTERNAL_SERVER_ERROR, "INTERNAL_ERROR"),
        };

        let body = serde_json::json!({
            "error": { "code": code, "message": self.to_string() }
        });

        (status, axum::Json(body)).into_response()
    }
}
```

## 九、幂等设计

### 9.1 幂等场景总览

testbed 的两个核心写操作需要幂等保护：

| 场景 | 触发方式 | 重复原因 | 幂等键 |
|------|---------|---------|--------|
| 测试执行（POST /jobs） | Agent 调用 device_run_test | Agent 超时重试 / 网络抖动 | `idempotency_key`（Agent 生成的 UUID） |
| 结果上报（POST platform /api/artifacts） | testbed 内部调用 | testbed 重启后重新上报 / 网络超时重试 | `job_id`（任务 ID 天然唯一） |

### 9.2 测试执行幂等

**问题**：Agent 调用 `device_run_test` 后网络超时，Agent 不知道任务是否已创建，于是重试。如果不做幂等，同一个测试会在设备上执行两次。

**方案**：请求级幂等键

```rust
// api/job.rs
//
// 创建测试任务 — 幂等保护。
//
// Agent 在请求中携带 idempotency_key（UUID），testbed 用它去重。
// 如果 idempotency_key 已存在，返回已有任务的状态，不重复创建。

#[derive(Debug, Deserialize)]
pub struct CreateJobRequest {
    pub device_id: String,
    pub artifact_id: String,
    pub test_script: String,           // YAML 内容
    pub idempotency_key: String,       // Agent 生成的 UUID，用于去重
    pub priority: Option<i32>,
}

pub async fn create_job(
    State(state): State<AppState>,
    Json(req): Json<CreateJobRequest>,
) -> Result<Json<TestJob>, TestbedError> {
    // ① 用 idempotency_key 查找已有任务
    if let Some(existing) = state.job_store.get_by_idempotency_key(&req.idempotency_key).await? {
        // 幂等命中：返回已有任务，不重复创建
        tracing::info!(
            idempotency_key = %req.idempotency_key,
            job_id = %existing.job_id,
            "idempotent hit: returning existing job"
        );
        return Ok(Json(existing));
    }

    // ② 正常创建任务
    let job = TestJob {
        job_id: Uuid::new_v4().to_string(),
        idempotency_key: req.idempotency_key.clone(),
        device_id: req.device_id,
        // ... 其他字段
    };

    state.job_store.insert(job.clone()).await?;
    state.scheduler.enqueue(job.clone()).await?;

    Ok(Json(job))
}
```

**存储结构**：

```rust
// job/queue.rs
//
// 任务存储 — 内存 HashMap，支持按 idempotency_key 查找。
//
// Phase 3 用内存存储（进程重启后丢失，可接受）。
// Phase 4 持久化到 SQLite/PostgreSQL。

pub struct JobStore {
    jobs: RwLock<HashMap<String, TestJob>>,              // job_id → TestJob
    idempotency_index: RwLock<HashMap<String, String>>,  // idempotency_key → job_id
}

impl JobStore {
    pub async fn insert(&self, job: TestJob) -> Result<(), TestbedError> {
        let mut jobs = self.jobs.write().await;
        let mut idx = self.idempotency_index.write().await;
        idx.insert(job.idempotency_key.clone(), job.job_id.clone());
        jobs.insert(job.job_id.clone(), job);
        Ok(())
    }

    pub async fn get_by_idempotency_key(&self, key: &str) -> Result<Option<TestJob>, TestbedError> {
        let idx = self.idempotency_index.read().await;
        if let Some(job_id) = idx.get(key) {
            let jobs = self.jobs.read().await;
            return Ok(jobs.get(job_id).cloned());
        }
        Ok(None)
    }
}
```

**幂等键过期**：idempotency_key 保留 1 小时后清理（防止内存无限增长）。超过 1 小时的重试视为新请求。

### 9.3 结果上报幂等

**问题**：testbed 将测试结果上报到 platform（POST /api/artifacts）时，如果网络超时，testbed 会重试。如果 platform 已经收到并存储了结果，重试会导致重复 artifact。

**方案**：以 `job_id` 作为天然幂等键

```rust
// result/reporter.rs
//
// 结果上报 — 以 job_id 作为幂等键。
//
// 上报时在 metadata 中携带 job_id，platform 端用 job_id 去重。
// 如果 platform 返回 409 Conflict（已存在），视为上报成功。

pub struct ResultReporter {
    platform_url: String,
    client: reqwest::Client,
    max_retries: u32,          // 最大重试次数（默认 3）
}

impl ResultReporter {
    pub async fn report(&self, result: &TestResult) -> Result<(), TestbedError> {
        let payload = serde_json::json!({
            "type": "device_test_report",
            "project_id": result.project_id,
            "name": format!("test-report-{}", result.job_id),
            "metadata": {
                "job_id": result.job_id,           // 幂等键
                "device_id": result.device_id,
                "device_model": result.device_model,
                "platform": result.platform,
                "status": result.status,
                "steps_total": result.steps_total,
                "steps_passed": result.steps_passed,
                "duration_ms": result.duration_ms,
            },
            "content": result,                     // 完整结果 JSON
        });

        let mut attempts = 0;
        loop {
            attempts += 1;
            let resp = self.client
                .post(format!("{}/api/artifacts", self.platform_url))
                .json(&payload)
                .send()
                .await;

            match resp {
                Ok(r) if r.status().is_success() => {
                    tracing::info!(job_id = %result.job_id, "result reported to platform");
                    return Ok(());
                }
                Ok(r) if r.status() == reqwest::StatusCode::CONFLICT => {
                    // 409 = 已存在，幂等成功
                    tracing::info!(job_id = %result.job_id, "result already reported (idempotent)");
                    return Ok(());
                }
                Ok(r) => {
                    tracing::warn!(job_id = %result.job_id, status = %r.status(), "report failed");
                }
                Err(e) => {
                    tracing::warn!(job_id = %result.job_id, error = %e, "report network error");
                }
            }

            if attempts >= self.max_retries {
                return Err(TestbedError::InternalError(
                    format!("failed to report result after {} attempts", self.max_retries)
                ));
            }

            // 指数退避：1s, 2s, 4s
            tokio::time::sleep(Duration::from_secs(1 << (attempts - 1))).await;
        }
    }
}
```

### 9.4 幂等设计总结

| 操作 | 幂等键 | 存储位置 | 过期策略 | 冲突处理 |
|------|--------|---------|---------|---------|
| POST /jobs | `idempotency_key`（请求参数） | testbed 内存（JobStore） | 1 小时后清理 | 返回已有任务 |
| POST platform /api/artifacts | `job_id`（metadata 字段） | platform 数据库 | 不过期（artifact 永久存储） | platform 返回 409，testbed 视为成功 |
